   141→/* ═══════════════════ Seeded Random Generators ═══════════════════ */
   142→
   143→function makeRng(seed: number) {
   144→  let s = seed
   145→  return () => { s = (s * 16807) % 2147483647; return s / 2147483647 }
   146→}

   149→  const r = makeRng(42)
   150→  const out: { x: number; y: number; w: number; h: number; s: number }[] = []
   151→  for (let i = 0; i < 800; i++) {
   152→    out.push({ x: (r() * WORLD_W) | 0, y: (r() * WORLD_H) | 0, w: (r() * 12 + 4) | 0, h: (r() * 8 + 3) | 0, s: (r() * 5) | 0 })
   153→  }
   154→  return out
   155→}

   243→/* ═══════════════════ Color Helpers ═══════════════════ */
   244→
   245→function hexToRgb(hex: string) {
   246→  const r = parseInt(hex.slice(1, 3), 16)
   247→  const g = parseInt(hex.slice(3, 5), 16)
   248→  const b = parseInt(hex.slice(5, 7), 16)
   249→  return { r, g, b }
   250→}
   251→
   252→function darken(hex: string, factor: number): string {
   253→  const { r, g, b } = hexToRgb(hex)
   254→  return `rgb(${Math.round(r * factor)},${Math.round(g * factor)},${Math.round(b * factor)})`
   255→}
   256→
   257→function lighten(hex: string, factor: number): string {
   258→  const { r, g, b } = hexToRgb(hex)
   259→  return `rgb(${Math.min(255, Math.round(r + (255 - r) * factor))},${Math.min(255, Math.round(g + (255 - g) * factor))},${Math.min(255, Math.round(b + (255 - b) * factor))})`
   260→}

   262→/* ═══════════════════ Rounded Rect Helper ═══════════════════ */
   263→
   264→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   265→  ctx.beginPath()
   266→  ctx.moveTo(x + r, y)
   267→  ctx.lineTo(x + w - r, y)
   268→  ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   269→  ctx.lineTo(x + w, y + h - r)
   270→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   271→  ctx.lineTo(x + r, y + h)
   272→  ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   273→  ctx.lineTo(x, y + r)
   274→  ctx.quadraticCurveTo(x, y, x + r, y)
   275→  ctx.closePath()
   276→}

   280→function drawStrokedText(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, strokeW: number = 2, strokeColor: string = '#000000', fillColor: string = '#FFFFFF') {
   281→  ctx.textAlign = 'center'
   282→  ctx.textBaseline = 'middle'
   283→  ctx.lineWidth = strokeW
   284→  ctx.strokeStyle = strokeColor
   285→  ctx.lineJoin = 'round'
   286→  ctx.miterLimit = 2
   287→  ctx.strokeText(text, x, y)
   288→  ctx.fillStyle = fillColor
   289→  ctx.fillText(text, x, y)
   290→}

   292→function drawGlowText(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, glowColor: string, fillColor: string = '#FFFFFF', fontSize: number = 10) {
   293→  ctx.textAlign = 'center'
   294→  ctx.textBaseline = 'middle'
   295→  ctx.font = `bold ${fontSize}px monospace`
   296→  // Glow layer
   297→  ctx.save()
   298→  ctx.shadowColor = glowColor
   299→  ctx.shadowBlur = 10
   300→  ctx.fillStyle = glowColor
   301→  ctx.fillText(text, x, y)
   302→  ctx.restore()
   303→  // Outline
   304→  ctx.font = `bold ${fontSize}px monospace`
   305→  ctx.textAlign = 'center'
   306→  ctx.textBaseline = 'middle'
   307→  ctx.lineWidth = 2.5
   308→  ctx.strokeStyle = 'rgba(0,0,0,0.8)'
   309→  ctx.lineJoin = 'round'
   310→  ctx.strokeText(text, x, y)
   311→  // Main fill
   312→  ctx.fillStyle = fillColor
   313→  ctx.fillText(text, x, y)
   314→}

   316→/* ═══════════════════ Background Drawing ═══════════════════ */

   318→function drawBackground(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number, time: number) {
   319→  // Deep space gradient
   320→  const grad = ctx.createLinearGradient(0, 0, 0, vh)
   321→  grad.addColorStop(0, '#030712')
   322→  grad.addColorStop(0.3, '#070d1f')
   323→  grad.addColorStop(0.6, '#0a1030')
   324→  grad.addColorStop(1, '#0e1545')
   325→  ctx.fillStyle = grad
   326→  ctx.fillRect(0, 0, vw, vh)
   327→
   328→  // Nebula clouds (parallax 0.12x)
   329→  for (const cloud of NEBULA_CLOUDS) {
   330→    const cx = (cloud.x * 0.12 - camX * 0.12) | 0
   331→    const cy = (cloud.y * 0.12 - camY * 0.12) | 0
   332→    if (cx + cloud.rx < -20 || cx - cloud.rx > vw + 20 || cy + cloud.ry < -20 || cy - cloud.ry > vh + 20) continue
   333→    const rgb = hexToRgb(cloud.color)
   334→    const alpha = cloud.alpha + Math.sin(time * 0.001 + cloud.x * 0.01) * 0.006
   335→    const cGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cloud.rx)
   336→    cGrad.addColorStop(0, `rgba(${rgb.r},${rgb.g},${rgb.b},${(alpha + 0.03).toFixed(3)})`)
   337→    cGrad.addColorStop(0.4, `rgba(${rgb.r},${rgb.g},${rgb.b},${alpha.toFixed(3)})`)
   338→    cGrad.addColorStop(1, `rgba(${rgb.r},${rgb.g},${rgb.b},0)`)
   339→    ctx.fillStyle = cGrad
   340→    ctx.beginPath()
   341→    ctx.ellipse(cx, cy, cloud.rx, cloud.ry, 0, 0, Math.PI * 2)
   342→    ctx.fill()
   343→  }
   344→
   345→  // Distant mountains (parallax 0.2x)
   346→  for (const mt of MOUNTAINS) {
   347→    const mx = (mt.x * 0.2 - camX * 0.2) | 0
   348→    const my = (vh * 0.6 - camY * 0.2) | 0
   349→    if (mx + mt.w < -10 || mx > vw + 10) continue
   350→    ctx.fillStyle = mt.color
   351→    ctx.beginPath()
   352→    ctx.moveTo(mx, my)
   353→    ctx.lineTo(mx + mt.w * 0.45, my - mt.h)
   354→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.9)
   355→    ctx.lineTo(mx + mt.w, my)
   356→    ctx.closePath()
   357→    ctx.fill()
   358→    // Mountain snow cap
   359→    ctx.fillStyle = 'rgba(0,229,255,0.04)'
   360→    ctx.beginPath()
   361→    ctx.moveTo(mx + mt.w * 0.4, my - mt.h * 0.7)
   362→    ctx.lineTo(mx + mt.w * 0.45, my - mt.h)
   363→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.9)
   364→    ctx.lineTo(mx + mt.w * 0.6, my - mt.h * 0.6)
   365→    ctx.closePath()
   366→    ctx.fill()
   367→  }
   368→
   369→  // Stars (parallax 0.35x)
   370→  for (const st of STARS) {
   371→    const sx = (st.x * 0.35 - camX * 0.35) | 0
   372→    const sy = (st.y * 0.35 - camY * 0.35) | 0
   373→    if (sx < -4 || sx > vw + 4 || sy < -4 || sy > vh + 4) continue
   374→    const brightness = (Math.sin(time * st.spd * 60 + st.b * 100) + 1) / 2
   375→    const alpha = 0.3 + brightness * 0.7
   376→    ctx.fillStyle = `rgba(255,255,255,${alpha.toFixed(2)})`
   377→    const s = st.size
   378→    if (s > 1.5 && brightness > 0.5) {
   379→      // Bright star with cross sparkle
   380→      ctx.fillRect(sx, sy, 2, 2)
   381→      ctx.fillStyle = `rgba(255,255,255,${(alpha * 0.25).toFixed(2)})`
   382→      ctx.fillRect(sx - 2, sy + 0.5, 6, 1)
   383→      ctx.fillRect(sx + 0.5, sy - 2, 1, 6)
   384→    } else {
   385→      ctx.fillRect(sx, sy, s > 1 ? 2 : 1, s > 1 ? 2 : 1)
   386→    }
   387→  }
   388→
   389→  // Atmospheric fog at bottom
   390→  const fogGrad = ctx.createLinearGradient(0, vh * 0.8, 0, vh)
   391→  fogGrad.addColorStop(0, 'rgba(10,15,40,0)')
   392→  fogGrad.addColorStop(1, 'rgba(10,15,40,0.3)')
   393→  ctx.fillStyle = fogGrad
   394→  ctx.fillRect(0, vh * 0.8, vw, vh * 0.2)
   395→}

   397→/* ═══════════════════ Ground Drawing ═══════════════════ */
   398→
   399→function drawGround(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number) {
   400→  // Grass patches with texture
   401→  for (const g of GRASS) {
   402→    const gx = (g.x - camX) | 0
   403→    const gy = (g.y - camY) | 0
   404→    if (gx < -16 || gx > vw + 16 || gy < -16 || gy > vh + 16) continue
   405→    ctx.fillStyle = GRASS_SHADES[g.s]
   406→    ctx.fillRect(gx, gy, g.w, g.h)
   407→    // Grass blade highlights
   408→    if (g.s % 2 === 0 && g.w > 5) {
   409→      ctx.fillStyle = 'rgba(0,229,255,0.025)'
   410→      ctx.fillRect(gx + 1, gy, 1, g.h)
   411→    }
   412→  }
   413→
   414→  // Paths with realistic pavement texture
   415→  for (const p of PATHS) {
   416→    const px = (p.x - camX) | 0
   417→    const py = (p.y - camY) | 0
   418→    if (px > vw + p.w || px + p.w < 0 || py > vh + p.h || py + p.h < 0) continue
   419→    // Base path with gradient
   420→    const pathGrad = ctx.createLinearGradient(px, py, px, py + p.h)
   421→    pathGrad.addColorStop(0, '#3d3d6b')
   422→    pathGrad.addColorStop(0.5, '#2c2c54')
   423→    pathGrad.addColorStop(1, '#3d3d6b')
   424→    ctx.fillStyle = pathGrad
   425→    ctx.fillRect(px, py, p.w, p.h)
   426→    // Top curb edge
   427→    ctx.fillStyle = '#4a4a7a'
