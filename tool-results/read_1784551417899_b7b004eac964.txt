    81→/* Seeded random for deterministic world generation */
    82→function makeRng(seed: number) {
    83→  let s = seed
    84→  return () => { s = (s * 16807) % 2147483647; return s / 2147483647 }
    85→}

    87→function genGrass() {
    88→  const r = makeRng(42)
    89→  const out: { x: number; y: number; w: number; h: number; s: number }[] = []
    90→  for (let i = 0; i < 500; i++) {
    91→    out.push({ x: (r() * WORLD_W) | 0, y: (r() * WORLD_H) | 0, w: (r() * 8 + 2) | 0, h: (r() * 5 + 2) | 0, s: (r() * 3) | 0 })
    92→  }

   150→function drawBackground(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number, time: number) {
   151→  ctx.fillStyle = '#1a1a2e'
   152→  ctx.fillRect(0, 0, vw, vh)
   153→
   154→  // Stars (parallax 0.4x)
   155→  for (const st of STARS) {
   156→    const sx = (st.x * 0.4 - camX * 0.4) | 0
   157→    const sy = (st.y * 0.4 - camY * 0.4) | 0
   158→    if (sx < -2 || sx > vw + 2 || sy < -2 || sy > vh + 2) continue
   159→    const brightness = (Math.sin(time * st.spd * 60 + st.b * 100) + 1) / 2
   160→    ctx.fillStyle = `rgba(255,255,255,${(0.2 + brightness * 0.8).toFixed(2)})`
   161→    ctx.fillRect(sx, sy, 1 + (brightness > 0.7 ? 1 : 0), 1)
   162→  }
   163→}

   277→function drawBuilding(ctx: CanvasRenderingContext2D, b: BuildingData, camX: number, camY: number, vw: number, vh: number, time: number, charX: number, charY: number) {
   278→  const bx = (b.x - camX) | 0
   279→  const by = (b.y - camY) | 0
   280→  if (bx + b.w < -20 || bx > vw + 20 || by + b.h < -30 || by > vh + 20) return
   281→
   282→  const bcx = b.x + b.w / 2
   283→  const bcy = b.y + b.h / 2
   284→  const dist = Math.hypot(charX + 8 - bcx, charY + 12 - bcy)
   285→  const isNear = dist < INTERACT_DIST
   286→
   287→  // Entrance glow when near
   288→  if (isNear) {
   289→    const alpha = Math.min(0.25, (INTERACT_DIST - dist) / INTERACT_DIST * 0.25)
   290→    ctx.fillStyle = `${b.glowColor}${Math.round(alpha * 255).toString(16).padStart(2, '0')}`
   291→    ctx.fillRect(bx + (b.w / 2 - 16) | 0, by + b.h - 8, 32, 8)
   292→  }
   293→
   294→  // Roof
   295→  ctx.fillStyle = b.color
   296→  ctx.fillRect(bx - 5, by - 14, b.w + 10, 16)
   297→
   298→  // Roof accent
   299→  ctx.fillStyle = b.glowColor
   300→  const roofPulse = (Math.sin(time * 0.004 + b.x * 0.1) + 1) / 2
   301→  ctx.globalAlpha = 0.3 + roofPulse * 0.4
   302→  ctx.fillRect(bx - 3, by - 12, b.w + 6, 2)
   303→  ctx.globalAlpha = 1
   304→
   305→  // Rooftop antenna
   306→  ctx.fillStyle = '#78909C'
   307→  ctx.fillRect(bx + (b.w / 2) | 0, by - 22, 2, 8)
   308→  ctx.fillStyle = b.glowColor
   309→  ctx.fillRect(bx + (b.w / 2 - 1) | 0, by - 24, 4, 3)
   310→
   311→  // Main structure
   312→  ctx.fillStyle = b.color
   313→  ctx.fillRect(bx, by, b.w, b.h)
   314→
   315→  // Building side shade
   316→  ctx.fillStyle = 'rgba(0,0,0,0.15)'
   317→  ctx.fillRect(bx + b.w - 4, by, 4, b.h)
   318→
   319→  // Windows
   320→  const windowGlow = (Math.sin(time * 0.002 + b.x * 0.05) + 1) / 2
   321→  for (let wy = by + 16; wy < by + b.h - 26; wy += 18) {
   322→    for (let wx = bx + 14; wx < bx + b.w - 18; wx += 24) {
   323→      if (windowGlow > 0.4) {
   324→        ctx.fillStyle = `rgba(255,249,196,${(windowGlow * 0.12).toFixed(2)})`
   325→        ctx.fillRect(wx - 1, wy - 1, 12, 12)
   326→      }
   327→      ctx.fillStyle = '#FFF9C4'
   328→      ctx.fillRect(wx, wy, 10, 10)
   329→      ctx.fillStyle = 'rgba(0,0,0,0.15)'
   330→      ctx.fillRect(wx + 4, wy, 2, 10)
   331→      ctx.fillRect(wx, wy + 4, 10, 2)
   332→    }
   333→  }
   334→
   335→  // Door
   336→  ctx.fillStyle = '#3E2723'
   337→  ctx.fillRect(bx + (b.w / 2 - 8) | 0, by + b.h - 20, 16, 20)
   338→  ctx.fillStyle = '#5D4037'
   339→  ctx.fillRect(bx + (b.w / 2 - 6) | 0, by + b.h - 18, 12, 16)
   340→  ctx.fillStyle = '#FFD740'
   341→  ctx.fillRect(bx + (b.w / 2 + 3) | 0, by + b.h - 10, 2, 2)
   342→
   343→  // Sign with glow
   344→  ctx.save()
   345→  ctx.shadowColor = b.glowColor
   346→  ctx.shadowBlur = 6
   347→  ctx.fillStyle = b.glowColor
   348→  ctx.font = 'bold 7px monospace'
   349→  ctx.textAlign = 'center'
   350→  ctx.textBaseline = 'bottom'
   351→  ctx.fillText(b.name, bx + (b.w / 2) | 0, by - 26)
   352→  ctx.restore()
   353→
   354→  // Bright text on top of glow
   355→  ctx.fillStyle = '#FFFFFF'
   356→  ctx.font = 'bold 7px monospace'
   357→  ctx.textAlign = 'center'
   358→  ctx.textBaseline = 'bottom'
   359→  ctx.fillText(b.name, bx + (b.w / 2) | 0, by - 26)
   360→}

   362→function drawNPC(ctx: CanvasRenderingContext2D, npc: NPCData, camX: number, camY: number, vw: number, vh: number, charX: number, charY: number, time: number) {
   363→  const nx = (npc.x - camX) | 0
   364→  const ny = (npc.y - camY) | 0
   365→  if (nx < -20 || nx > vw + 20 || ny < -40 || ny > vh + 20) return
   366→
   367→  const dist = Math.hypot(npc.x - charX - 8, npc.y - charY - 12)
   368→
   369→  if (npc.type === 'robot') {
   370→    // Body
   371→    ctx.fillStyle = npc.color
   372→    ctx.fillRect(nx - 5, ny - 12, 10, 12)
   373→    // Head
   374→    ctx.fillStyle = '#B0BEC5'
   375→    ctx.fillRect(nx - 4, ny - 20, 8, 8)
   376→    // Eyes
   377→    const blink = Math.sin(time * 0.005) > 0.92 ? 0 : 1
   378→    ctx.fillStyle = '#F44336'
   379→    ctx.fillRect(nx - 2, ny - 17, 2, 2 * blink)
   380→    ctx.fillRect(nx + 1, ny - 17, 2, 2 * blink)
   381→    // Antenna
   382→    ctx.fillStyle = npc.color
   383→    ctx.fillRect(nx, ny - 23, 1, 3)
   384→    ctx.fillRect(nx - 1, ny - 25, 3, 2)
   385→    // Hover bob
   386→    const hb = Math.sin(time * 0.006 + npc.x) * 1.5
   387→    if (Math.abs(hb) > 0.5) {
   388→      ctx.fillStyle = 'rgba(0,0,0,0.15)'
   389→      ctx.fillRect(nx - 4, ny + 1, 8, 2)
   390→    }
   391→  } else {
   392→    // Lab coat body
   393→    ctx.fillStyle = '#ECEFF1'
   394→    ctx.fillRect(nx - 5, ny - 12, 10, 14)
   395→    // Head
   396→    ctx.fillStyle = '#FFCCBC'
   397→    ctx.fillRect(nx - 4, ny - 20, 8, 8)
   398→    // Hair
   399→    ctx.fillStyle = '#5D4037'
   400→    ctx.fillRect(nx - 4, ny - 22, 8, 4)
   401→    // Glasses
   402→    ctx.fillStyle = '#263238'
   403→    ctx.fillRect(nx - 3, ny - 17, 3, 2)
   404→    ctx.fillRect(nx + 1, ny - 17, 3, 2)
   405→    ctx.fillRect(nx, ny - 17, 1, 2)
   406→    // Legs
   407→    ctx.fillStyle = '#455A64'
   408→    ctx.fillRect(nx - 3, ny + 2, 3, 4)
   409→    ctx.fillRect(nx + 1, ny + 2, 3, 4)
   410→  }
   411→
   412→  // Speech bubble
   413→  if (dist < NPC_DIST) {
   414→    const alpha = Math.min(1, (NPC_DIST - dist) / 20)
   415→    ctx.globalAlpha = alpha
   416→    const text = npc.message
   417→    ctx.font = '5px monospace'
   418→    const tw = ctx.measureText(text).width
   419→    const bw = tw + 8
   420→    const bh = 11
   421→    const bubX = (nx - bw / 2) | 0
   422→    const bubY = ny - (npc.type === 'robot' ? 38 : 38)
   423→
   424→    ctx.fillStyle = '#FFFFFF'
   425→    ctx.fillRect(bubX, bubY, bw, bh)
   426→    // Tail
   427→    ctx.fillRect(nx - 1, bubY + bh, 2, 3)
   428→    // Border
   429→    ctx.strokeStyle = '#000000'
   430→    ctx.lineWidth = 0.5
   431→    ctx.strokeRect(bubX, bubY, bw, bh)
   432→    // Text
   433→    ctx.fillStyle = '#212121'
   434→    ctx.textAlign = 'center'
   435→    ctx.textBaseline = 'middle'
   436→    ctx.fillText(text, nx, bubY + bh / 2 + 1)
   437→    ctx.globalAlpha = 1
   438→  }
   439→}

   441→function drawCharacter(
   442→  ctx: CanvasRenderingContext2D,
   443→  screenX: number,
   444→  screenY: number,
   445→  dir: string,
   446→  frame: number,
   447→  isMoving: boolean,
   448→  idleBob: number,
   449→  time: number
   450→) {
   451→  const bob = isMoving ? 0 : Math.round(Math.sin(idleBob * 0.12) * 1.5)
   452→  const walkBob = isMoving ? Math.round(Math.sin(time * 0.018) * 1) : 0
   453→
   454→  ctx.save()
   455→  ctx.translate(screenX | 0, (screenY + bob) | 0)
   456→
   457→  // Shadow
   458→  ctx.fillStyle = 'rgba(0,0,0,0.22)'
   459→  ctx.fillRect(1, 22, 14, 4)
   460→
   461→  // Legs
   462→  ctx.fillStyle = '#006064'
   463→  if (!isMoving) {
   464→    ctx.fillRect(3, 16, 4, 8)
   465→    ctx.fillRect(9, 16, 4, 8)
   466→  } else if (frame === 1) {
   467→    ctx.fillRect(2, 16, 4, 8)
   468→    ctx.fillRect(10, 18, 4, 6)
   469→  } else if (frame === 3) {
   470→    ctx.fillRect(4, 18, 4, 6)
   471→    ctx.fillRect(8, 16, 4, 8)
   472→  } else {
   473→    ctx.fillRect(3, 17, 4, 7)

   475→  }

   477→  // Body

   479→  ctx.fillRect(2, 8 + walkBob, 12, 8)
   480→  // Body stripe
   481→  ctx.fillStyle = '#00BCD4'
   482→  ctx.fillRect(2, 12 + walkBob, 12, 1)
   483→
   484→  // Arms

   486→  ctx.fillStyle = '#00BCD4'
   487→  ctx.fillRect(-1, 9 + walkBob + armSwing, 3, 5)
   488→  ctx.fillRect(14, 9 + walkBob - armSwing, 3, 5)
   489→
   490→  // Head

   492→  ctx.fillRect(3, 0, 10, 8)
   493→
   494→  // Hair

   496→  ctx.fillRect(3, 0, 10, 3)
   497→  if (dir === 'left') ctx.fillRect(3, 0, 2, 5)
   498→  else if (dir === 'right') ctx.fillRect(11, 0, 2, 5)
   499→
   500→  // Eyes
   501→  ctx.fillStyle = '#212121'
   502→  if (dir === 'down') {
   503→    ctx.fillRect(5, 4, 2, 2)
   504→    ctx.fillRect(9, 4, 2, 2)
   505→    ctx.fillRect(7, 6, 2, 1)
   506→  } else if (dir === 'up') {
   507→    ctx.fillStyle = '#4E342E'
   508→    ctx.fillRect(3, 2, 10, 2)
   509→  } else if (dir === 'left') {
   510→    ctx.fillRect(4, 4, 2, 2)
   511→    ctx.fillRect(7, 4, 2, 2)
   512→    ctx.fillRect(5, 6, 2, 1)
   513→  } else {
   514→    ctx.fillRect(7, 4, 2, 2)
   515→    ctx.fillRect(10, 4, 2, 2)
   516→    ctx.fillRect(9, 6, 2, 1)
   517→  }
   518→
   519→  ctx.restore()
   520→}

   522→function drawMinimap(ctx: CanvasRenderingContext2D, vw: number, vh: number, charX: number, charY: number, camX: number, camY: number) {
   523→  const mw = 90
   524→  const mh = 68
   525→  const mx = vw - mw - 8
   526→  const my = 8
   527→  const sx = mw / WORLD_W
   528→  const sy = mh / WORLD_H
   529→
   530→  ctx.fillStyle = 'rgba(0,0,0,0.55)'
   531→  ctx.fillRect(mx - 1, my - 1, mw + 2, mh + 2)
   532→  ctx.fillStyle = 'rgba(26,26,46,0.8)'
   533→  ctx.fillRect(mx, my, mw, mh)
   534→
   535→  // Paths
   536→  ctx.fillStyle = 'rgba(44,44,84,0.7)'
   537→  for (const p of PATHS) {
   538→    ctx.fillRect(mx + p.x * sx, my + p.y * sy, Math.max(p.w * sx, 1), Math.max(p.h * sy, 1))
   539→  }
   540→
   541→  // Buildings
   542→  for (const b of BUILDINGS) {
   543→    ctx.fillStyle = b.glowColor
   544→    ctx.fillRect(mx + b.x * sx, my + b.y * sy, Math.max(b.w * sx, 3), Math.max(b.h * sy, 2))
   545→  }
   546→
   547→  // Viewport rect
   548→  ctx.strokeStyle = 'rgba(255,255,255,0.3)'
   549→  ctx.lineWidth = 0.5
   550→  ctx.strokeRect(mx + camX * sx, my + camY * sy, vw * sx, vh * sy)
   551→
   552→  // Character dot
   553→  ctx.fillStyle = '#00E5FF'
   554→  ctx.fillRect(mx + charX * sx - 1, my + charY * sy - 1, 3, 3)
   555→
   556→  // Border
   557→  ctx.strokeStyle = 'rgba(255,255,255,0.25)'
   558→  ctx.lineWidth = 1
   559→  ctx.strokeRect(mx - 1, my - 1, mw + 2, mh + 2)
   560→}

   562→function drawPrompt(ctx: CanvasRenderingContext2D, building: BuildingData | null, camX: number, camY: number, isMobile: boolean, time: number) {
   563→  if (!building) return
   564→
   565→  const bx = (building.x + building.w / 2 - camX) | 0
   566→  const by = (building.y + building.h + 14 - camY) | 0
   567→  const text = isMobile ? 'Tap [E] to enter' : 'Press E to enter'
   568→
   569→  ctx.font = '6px monospace'
   570→  const tw = ctx.measureText(text).width
   571→  const pw = tw + 10
   572→  const ph = 13
   573→
   574→  // Pulsing alpha
   575→  const alpha = 0.7 + Math.sin(time * 0.006) * 0.3
   576→  ctx.globalAlpha = alpha
   577→
   578→  ctx.fillStyle = 'rgba(0,0,0,0.75)'
   579→  ctx.fillRect(bx - pw / 2, by, pw, ph)
   580→
   581→  ctx.fillStyle = building.glowColor
   582→  ctx.textAlign = 'center'
   583→  ctx.textBaseline = 'middle'
   584→  ctx.fillText(text, bx, by + ph / 2)
   585→
   586→  ctx.globalAlpha = 1
   587→}

   591→export default function PixelWorld({ onBuildingClick }: PixelWorldProps) {
   592→  const canvasRef = useRef<HTMLCanvasElement>(null)
   593→  const containerRef = useRef<HTMLDivElement>(null)
   594→  const offscreenRef = useRef<HTMLCanvasElement | null>(null)
   595→  const animRef = useRef(0)
   596→  const timeRef = useRef(0)
   597→
   598→  const stateRef = useRef({
   599→    charX: WORLD_W / 2 - 8,
   600→    charY: WORLD_H / 2 - 12,
   601→    dir: 'down' as 'left' | 'right' | 'up' | 'down',
   602→    frame: 0,
   603→    frameTimer: 0,
   604→    isMoving: false,
   605→    camX: 0,
   606→    camY: 0,
   607→    nearBuilding: null as string | null,
   608→    idleBob: 0,
   609→    targetX: -1,
   610→    targetY: -1,
   611→  })
   612→
   613→  const keysRef = useRef(new Set<string>())
   614→  const joystickRef = useRef({ active: false, dx: 0, dy: 0 })
   615→  const thumbRef = useRef<HTMLDivElement>(null)
   616→  const particlesRef = useRef<Particle[]>([])
   617→
   618→  const [isMobile, setIsMobile] = useState(false)
   619→  const [nearBuilding, setNearBuilding] = useState<string | null>(null)
   620→  const [showHelp, setShowHelp] = useState(true)
   621→
   622→  // Detect mobile
   623→  useEffect(() => {
   624→    const check = () => setIsMobile('ontouchstart' in window || navigator.maxTouchPoints > 0)
   625→    check()
   626→    const onResize = () => check()
   627→    window.addEventListener('resize', onResize)
   628→    return () => window.removeEventListener('resize', onResize)
   629→  }, [])
   630→
   631→  // Hide help
   632→  useEffect(() => {
   633→    const t = setTimeout(() => setShowHelp(false), 6000)
   634→    return () => clearTimeout(t)
   635→  }, [])
   636→
   637→  // Find nearest building
   638→  const findNearBuilding = useCallback((cx: number, cy: number): BuildingData | null => {
   639→    let nearest: BuildingData | null = null
   640→    let minDist = INTERACT_DIST
   641→    for (const b of BUILDINGS) {
   642→      const d = Math.hypot(cx - (b.x + b.w / 2), cy - (b.y + b.h / 2))
   643→      if (d < minDist) { minDist = d; nearest = b }
   644→    }
   645→    return nearest
   646→  }, [])
   647→
   648→  // Interact
   649→  const interact = useCallback(() => {
   650→    const s = stateRef.current
   651→    const b = findNearBuilding(s.charX + 8, s.charY + 12)
   652→    if (b) onBuildingClick(b.id)
   653→  }, [onBuildingClick, findNearBuilding])
   654→
   655→  // Keyboard
   656→  useEffect(() => {
   657→    const onDown = (e: KeyboardEvent) => {
   658→      const k = e.key.toLowerCase()
   659→      keysRef.current.add(k)
   660→      if (k === 'e' || k === 'enter') interact()
   661→    }
   662→    const onUp = (e: KeyboardEvent) => keysRef.current.delete(e.key.toLowerCase())
   663→    window.addEventListener('keydown', onDown)
   664→    window.addEventListener('keyup', onUp)
   665→    return () => { window.removeEventListener('keydown', onDown); window.removeEventListener('keyup', onUp) }
   666→  }, [interact])
   667→
   668→  // Canvas click-to-move
   669→  const handleCanvasClick = useCallback((e: React.MouseEvent<HTMLCanvasElement>) => {
   670→    const canvas = canvasRef.current
   671→    if (!canvas) return
   672→    const rect = canvas.getBoundingClientRect()
   673→    const s = stateRef.current
   674→    const offscreen = offscreenRef.current
   675→    if (!offscreen) return
   676→    const scaleX = offscreen.width / rect.width
   677→    const scaleY = offscreen.height / rect.height
   678→    s.targetX = (e.clientX - rect.left) * scaleX + s.camX
   679→    s.targetY = (e.clientY - rect.top) * scaleY + s.camY
   680→  }, [])
   681→
   682→  // Touch tap-to-move on canvas
   683→  const handleCanvasTouch = useCallback((e: React.TouchEvent<HTMLCanvasElement>) => {
   684→    const canvas = canvasRef.current
   685→    if (!canvas) return
   686→    const rect = canvas.getBoundingClientRect()
   687→    const touch = e.changedTouches[0]
   688→    if (touch.clientY > rect.bottom - 130) return
   689→    const s = stateRef.current
   690→    const offscreen = offscreenRef.current
   691→    if (!offscreen) return
   692→    const scaleX = offscreen.width / rect.width
   693→    const scaleY = offscreen.height / rect.height
   694→    s.targetX = (touch.clientX - rect.left) * scaleX + s.camX
   695→    s.targetY = (touch.clientY - rect.top) * scaleY + s.camY
   696→  }, [])
   697→
   698→  // Joystick handlers
   699→  const handleJoyStart = useCallback((e: React.TouchEvent) => {
   700→    e.preventDefault(); e.stopPropagation()
   701→    joystickRef.current.active = true
   702→    joystickRef.current.dx = 0; joystickRef.current.dy = 0
   703→    stateRef.current.targetX = -1; stateRef.current.targetY = -1
   704→    if (thumbRef.current) thumbRef.current.style.transform = 'translate(0px, 0px)'
   705→  }, [])
   706→
   707→  const handleJoyMove = useCallback((e: React.TouchEvent) => {
   708→    e.preventDefault(); e.stopPropagation()
   709→    if (!joystickRef.current.active) return
   710→    const touch = e.touches[0]
   711→    const base = document.getElementById('joystick-base')
   712→    if (!base) return
   713→    const rect = base.getBoundingClientRect()
   714→    const cx = rect.left + rect.width / 2
   715→    const cy = rect.top + rect.height / 2
   716→    let dx = touch.clientX - cx
   717→    let dy = touch.clientY - cy
   718→    const maxR = rect.width / 2 - 20
   719→    const dist = Math.hypot(dx, dy)
   720→    if (dist > maxR) { dx = dx / dist * maxR; dy = dy / dist * maxR }
   721→    joystickRef.current.dx = dx / maxR
   722→    joystickRef.current.dy = dy / maxR
   723→    if (thumbRef.current) thumbRef.current.style.transform = `translate(${dx}px, ${dy}px)`
   724→  }, [])
   725→
   726→  const handleJoyEnd = useCallback((e: React.TouchEvent) => {
   727→    e.preventDefault()
   728→    joystickRef.current.active = false
   729→    joystickRef.current.dx = 0; joystickRef.current.dy = 0
   730→    if (thumbRef.current) thumbRef.current.style.transform = 'translate(0px, 0px)'
   731→  }, [])
   732→
   733→  // ── Main Game Loop Effect ──
   734→  useEffect(() => {
   735→    const offscreen = document.createElement('canvas')
   736→    offscreenRef.current = offscreen
   737→
   738→    const resize = () => {
   739→      const canvas = canvasRef.current
   740→      const container = containerRef.current
   741→      if (!canvas || !container) return
   742→      const cw = container.clientWidth
   743→      const ch = container.clientHeight
   744→      canvas.width = cw
   745→      canvas.height = ch
   746→      offscreen.width = Math.ceil(cw / PIXEL_SCALE)
   747→      offscreen.height = Math.ceil(ch / PIXEL_SCALE)
   748→    }
   749→    resize()
   750→    window.addEventListener('resize', resize)
   751→
   752→    const loop = () => {
   753→      const canvas = canvasRef.current
   754→      const ctx = canvas?.getContext('2d')
   755→      const offCtx = offscreen?.getContext('2d')
   756→      if (!canvas || !ctx || !offCtx) { animRef.current = requestAnimationFrame(loop); return }
   757→
   758→      const s = stateRef.current
   759→      const keys = keysRef.current
   760→      const joy = joystickRef.current
   761→      timeRef.current++
   762→      const time = timeRef.current
   763→      const vw = offscreen.width
   764→      const vh = offscreen.height
   765→
   766→      // ── Input Processing ──
   767→      let dx = 0, dy = 0
   768→      if (keys.has('arrowleft') || keys.has('a')) dx -= 1
   769→      if (keys.has('arrowright') || keys.has('d')) dx += 1
   770→      if (keys.has('arrowup') || keys.has('w')) dy -= 1
   771→      if (keys.has('arrowdown') || keys.has('s')) dy += 1
   772→
   773→      if (joy.active && (Math.abs(joy.dx) > 0.15 || Math.abs(joy.dy) > 0.15)) {
   774→        dx += joy.dx
   775→        dy += joy.dy
   776→      }
   777→
   778→      // Click/tap to move
   779→      if (s.targetX >= 0 && s.targetY >= 0 && !joy.active && dx === 0 && dy === 0) {
   780→        const tdx = s.targetX - (s.charX + 8)
   781→        const tdy = s.targetY - (s.charY + 12)
   782→        const tdist = Math.hypot(tdx, tdy)
   783→        if (tdist > CHAR_SPEED + 1) {
   784→          dx += tdx / tdist
   785→          dy += tdy / tdist
   786→        } else {
   787→          s.targetX = -1; s.targetY = -1
   788→        }
   789→      } else if (dx !== 0 || dy !== 0) {
   790→        s.targetX = -1; s.targetY = -1
   791→      }
   792→
   793→      // Normalize
   794→      const len = Math.hypot(dx, dy)
   795→      if (len > 0.01) {
   796→        const speed = CHAR_SPEED
   797→        dx = dx / len * speed
   798→        dy = dy / len * speed
   799→        s.isMoving = true
   800→
   801→        if (Math.abs(dx) > Math.abs(dy)) s.dir = dx > 0 ? 'right' : 'left'
   802→        else s.dir = dy > 0 ? 'down' : 'up'
   803→
   804→        s.frameTimer++
   805→        if (s.frameTimer >= 7) { s.frameTimer = 0; s.frame = (s.frame + 1) % 4 }
   806→
   807→        // Dust particles
   808→        if (time % 8 === 0) {
   809→          particlesRef.current.push({
   810→            x: s.charX + 8 + (Math.random() - 0.5) * 6,
   811→            y: s.charY + 23,
   812→            vx: (Math.random() - 0.5) * 0.4,
   813→            vy: -Math.random() * 0.3 - 0.1,
   814→            life: 18, maxLife: 18,
   815→            color: '#4a4a6a', size: 2,
   816→          })
   817→        }
   818→      } else {
   819→        s.isMoving = false
   820→        s.frame = 0; s.frameTimer = 0
   821→        s.idleBob++
   822→      }
   823→
   824→      // Move character
   825→      s.charX = Math.max(0, Math.min(WORLD_W - 16, s.charX + dx))
   826→      s.charY = Math.max(0, Math.min(WORLD_H - 24, s.charY + dy))
   827→
   828→      // Camera (smooth follow)
   829→      const targetCamX = s.charX + 8 - vw / 2
   830→      const targetCamY = s.charY + 12 - vh / 2
   831→      const clampedX = Math.max(0, Math.min(WORLD_W - vw, targetCamX))
   832→      const clampedY = Math.max(0, Math.min(WORLD_H - vh, targetCamY))
   833→      s.camX += (clampedX - s.camX) * CAM_LERP
   834→      s.camY += (clampedY - s.camY) * CAM_LERP
   835→
   836→      // Near building detection
   837→      const nearB = findNearBuilding(s.charX + 8, s.charY + 12)
   838→      const newNear = nearB?.id ?? null
   839→      if (newNear !== s.nearBuilding) {
   840→        s.nearBuilding = newNear
   841→        setNearBuilding(newNear)
   842→      }
   843→
   844→      // Update particles
   845→      particlesRef.current = particlesRef.current.filter(p => {
   846→        p.x += p.vx; p.y += p.vy; p.life--
   847→        return p.life > 0
   848→      })
   849→
   850→      // ── Render ──
   851→      offCtx.imageSmoothingEnabled = false
   852→      const camX = s.camX
   853→      const camY = s.camY
   854→
   855→      drawBackground(offCtx, vw, vh, camX, camY, time)
   856→      drawGround(offCtx, vw, vh, camX, camY)
   857→      drawFlowers(offCtx, vw, vh, camX, camY)
   858→
   859→      // Trees (behind buildings/character)
   860→      for (const tree of TREES) drawTree(offCtx, tree, camX, camY, vw, vh)
   861→
   862→      // Lamps
   863→      for (const lamp of LAMPS) drawLamp(offCtx, lamp, camX, camY, vw, vh, time)
   864→
   865→      // Center monument
   866→      drawCenterMonument(offCtx, camX, camY, vw, vh, time)
   867→
   868→      // Y-sorted entities: buildings + NPCs + character
   869→      const entities: { y: number; draw: () => void }[] = []
   870→
   871→      for (const b of BUILDINGS) {
   872→        const bRef = b
   873→        entities.push({ y: b.y + b.h, draw: () => drawBuilding(offCtx, bRef, camX, camY, vw, vh, time, s.charX, s.charY) })
   874→      }
   875→
   876→      for (const npc of NPCS) {
   877→        const nRef = npc
   878→        entities.push({ y: npc.y + (npc.type === 'robot' ? 0 : 6), draw: () => drawNPC(offCtx, nRef, camX, camY, vw, vh, s.charX, s.charY, time) })
   879→      }
   880→
   881→      entities.push({
   882→        y: s.charY + 24,
   883→        draw: () => drawCharacter(offCtx, s.charX - camX, s.charY - camY, s.dir, s.frame, s.isMoving, s.idleBob, time),
   884→      })
   885→
   886→      entities.sort((a, b) => a.y - b.y)
   887→      for (const e of entities) e.draw()
   888→
   889→      // Particles
   890→      for (const p of particlesRef.current) {
   891→        const px = (p.x - camX) | 0
   892→        const py = (p.y - camY) | 0
   893→        offCtx.globalAlpha = p.life / p.maxLife * 0.6
   894→        offCtx.fillStyle = p.color
   895→        offCtx.fillRect(px, py, p.size, p.size)
   896→      }
   897→      offCtx.globalAlpha = 1
   898→
   899→      // UI
   900→      drawPrompt(offCtx, nearB, camX, camY, isMobile, time)
   901→      drawMinimap(offCtx, vw, vh, s.charX, s.charY, camX, camY)
   902→
   903→      // ── Blit to display ──
   904→      ctx.imageSmoothingEnabled = false
   905→      ctx.clearRect(0, 0, canvas.width, canvas.height)
   906→      ctx.drawImage(offscreen, 0, 0, canvas.width, canvas.height)
   907→
   908→      animRef.current = requestAnimationFrame(loop)
   909→    }
   910→
   911→    animRef.current = requestAnimationFrame(loop)
   912→
   913→    return () => {
   914→      cancelAnimationFrame(animRef.current)
   915→      window.removeEventListener('resize', resize)
   916→    }
   917→  }, [isMobile, findNearBuilding])
   918→
   919→  return (
   920→    <div ref={containerRef} className="relative w-full h-full overflow-hidden bg-[#1a1a2e]" style={{ minHeight: '300px' }}>
   921→      <canvas
   922→        ref={canvasRef}
   923→        className="w-full h-full block cursor-crosshair"
   924→        style={{ imageRendering: 'pixelated' }}
   925→        onClick={handleCanvasClick}
   926→        onTouchEnd={handleCanvasTouch}
   927→      />
   928→
   929→      {/* Mobile virtual joystick */}
   930→      {isMobile && (
   931→        <div
   932→          className="absolute bottom-6 left-6 touch-none select-none"
   933→          onTouchStart={handleJoyStart}
   934→          onTouchMove={handleJoyMove}
   935→          onTouchEnd={handleJoyEnd}
   936→        >
   937→          <div
   938→            id="joystick-base"
   939→            className="w-28 h-28 rounded-full bg-white/10 border-2 border-white/20 flex items-center justify-center backdrop-blur-sm"
   940→          >
   941→            <div
   942→              ref={thumbRef}
   943→              className="w-12 h-12 rounded-full bg-white/30 border-2 border-white/40"
   944→              style={{ transition: 'transform 0.05s ease-out' }}
   945→            />
   946→          </div>
   947→        </div>
   948→      )}
   949→
   950→      {/* Mobile action button */}
   951→      {isMobile && (
   952→        <button
   953→          className={`absolute bottom-8 right-8 w-16 h-16 rounded-full flex items-center justify-center text-white font-bold text-xl transition-all duration-200 touch-none select-none ${
   954→            nearBuilding
   955→              ? 'bg-white/25 border-2 border-white/60 scale-110 shadow-lg shadow-white/20'
   956→              : 'bg-white/10 border border-white/20 scale-100 opacity-40'
   957→          }`}
   958→          onTouchStart={(e) => { e.preventDefault(); interact() }}
   959→          aria-label="Interact with building"
   960→        >
   961→          E
   962→        </button>
   963→      )}
   964→
   965→      {/* Help text */}
   966→      {showHelp && (
