   131→/* ═══════════════════ Seeded Random Generators ═══════════════════ */
   132→
   133→function makeRng(seed: number) {
   134→  let s = seed
   135→  return () => { s = (s * 16807) % 2147483647; return s / 2147483647 }
   136→}

   138→function genGrass() {
   139→  const r = makeRng(42)
   140→  const out: { x: number; y: number; w: number; h: number; s: number }[] = []
   141→  for (let i = 0; i < 800; i++) {
   142→    out.push({ x: (r() * WORLD_W) | 0, y: (r() * WORLD_H) | 0, w: (r() * 12 + 4) | 0, h: (r() * 8 + 3) | 0, s: (r() * 5) | 0 })
   143→  }
   144→  return out
   145→}

   233→/* ═══════════════════ Color Helpers ═══════════════════ */

   235→function hexToRgb(hex: string) {
   236→  const r = parseInt(hex.slice(1, 3), 16)
   237→  const g = parseInt(hex.slice(3, 5), 16)
   238→  const b = parseInt(hex.slice(5, 7), 16)
   239→  return { r, g, b }
   240→}
   241→
   242→function darken(hex: string, factor: number): string {
   243→  const { r, g, b } = hexToRgb(hex)
   244→  return `rgb(${Math.round(r * factor)},${Math.round(g * factor)},${Math.round(b * factor)})`
   245→}
   246→
   247→function lighten(hex: string, factor: number): string {
   248→  const { r, g, b } = hexToRgb(hex)
   249→  return `rgb(${Math.min(255, Math.round(r + (255 - r) * factor))},${Math.min(255, Math.round(g + (255 - g) * factor))},${Math.min(255, Math.round(b + (255 - b) * factor))})`
   250→}

   252→/* ═══════════════════ Rounded Rect Helper ═══════════════════ */

   254→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   255→  ctx.beginPath()
   256→  ctx.moveTo(x + r, y)
   257→  ctx.lineTo(x + w - r, y)
   258→  ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   259→  ctx.lineTo(x + w, y + h - r)
   260→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   261→  ctx.lineTo(x + r, y + h)
   262→  ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   263→  ctx.lineTo(x, y + r)
   264→  ctx.quadraticCurveTo(x, y, x + r, y)
   265→  ctx.closePath()
   266→}

   270→function drawStrokedText(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, strokeW: number = 2, strokeColor: string = '#000000', fillColor: string = '#FFFFFF') {
   271→  ctx.textAlign = 'center'

   275→  ctx.lineJoin = 'round'
   276→  ctx.miterLimit = 2

   278→  ctx.fillStyle = fillColor
   279→  ctx.fillText(text, x, y)
   280→}

   282→function drawGlowText(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, glowColor: string, fillColor: string = '#FFFFFF', fontSize: number = 10) {
   283→  ctx.textAlign = 'center'
   284→  ctx.textBaseline = 'middle'
   285→  ctx.font = `bold ${fontSize}px monospace`
   286→  // Glow layer
   287→  ctx.save()
   288→  ctx.shadowColor = glowColor
   289→  ctx.shadowBlur = 10
   290→  ctx.fillStyle = glowColor
   291→  ctx.fillText(text, x, y)
   292→  ctx.restore()
   293→  // Outline
   294→  ctx.font = `bold ${fontSize}px monospace`
   295→  ctx.textAlign = 'center'
   296→  ctx.textBaseline = 'middle'
   297→  ctx.lineWidth = 2.5
   298→  ctx.strokeStyle = 'rgba(0,0,0,0.8)'
   299→  ctx.lineJoin = 'round'
   300→  ctx.strokeText(text, x, y)
   301→  // Main fill
   302→  ctx.fillStyle = fillColor
   303→  ctx.fillText(text, x, y)
   304→}

   308→function drawBackground(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number, time: number) {
   309→  // Deep space gradient
   310→  const grad = ctx.createLinearGradient(0, 0, 0, vh)
   311→  grad.addColorStop(0, '#030712')
   312→  grad.addColorStop(0.3, '#070d1f')
   313→  grad.addColorStop(0.6, '#0a1030')
   314→  grad.addColorStop(1, '#0e1545')
   315→  ctx.fillStyle = grad
   316→  ctx.fillRect(0, 0, vw, vh)
   317→
   318→  // Nebula clouds (parallax 0.12x)
   319→  for (const cloud of NEBULA_CLOUDS) {
   320→    const cx = (cloud.x * 0.12 - camX * 0.12) | 0
   321→    const cy = (cloud.y * 0.12 - camY * 0.12) | 0
   322→    if (cx + cloud.rx < -20 || cx - cloud.rx > vw + 20 || cy + cloud.ry < -20 || cy - cloud.ry > vh + 20) continue
   323→    const rgb = hexToRgb(cloud.color)
   324→    const alpha = cloud.alpha + Math.sin(time * 0.001 + cloud.x * 0.01) * 0.006
   325→    const cGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cloud.rx)
   326→    cGrad.addColorStop(0, `rgba(${rgb.r},${rgb.g},${rgb.b},${(alpha + 0.03).toFixed(3)})`)
   327→    cGrad.addColorStop(0.4, `rgba(${rgb.r},${rgb.g},${rgb.b},${alpha.toFixed(3)})`)
   328→    cGrad.addColorStop(1, `rgba(${rgb.r},${rgb.g},${rgb.b},0)`)
   329→    ctx.fillStyle = cGrad
   330→    ctx.beginPath()
   331→    ctx.ellipse(cx, cy, cloud.rx, cloud.ry, 0, 0, Math.PI * 2)
   332→    ctx.fill()
   333→  }
   334→
   335→  // Distant mountains (parallax 0.2x)
   336→  for (const mt of MOUNTAINS) {
   337→    const mx = (mt.x * 0.2 - camX * 0.2) | 0
   338→    const my = (vh * 0.6 - camY * 0.2) | 0
   339→    if (mx + mt.w < -10 || mx > vw + 10) continue
   340→    ctx.fillStyle = mt.color
   341→    ctx.beginPath()
   342→    ctx.moveTo(mx, my)
   343→    ctx.lineTo(mx + mt.w * 0.45, my - mt.h)
   344→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.9)
   345→    ctx.lineTo(mx + mt.w, my)
   346→    ctx.closePath()
   347→    ctx.fill()
   348→    // Mountain snow cap
   349→    ctx.fillStyle = 'rgba(0,229,255,0.04)'
   350→    ctx.beginPath()
   351→    ctx.moveTo(mx + mt.w * 0.4, my - mt.h * 0.7)
   352→    ctx.lineTo(mx + mt.w * 0.45, my - mt.h)
   353→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.9)
   354→    ctx.lineTo(mx + mt.w * 0.6, my - mt.h * 0.6)
   355→    ctx.closePath()
   356→    ctx.fill()
   357→  }
   358→
   359→  // Stars (parallax 0.35x)
   360→  for (const st of STARS) {
   361→    const sx = (st.x * 0.35 - camX * 0.35) | 0
   362→    const sy = (st.y * 0.35 - camY * 0.35) | 0
   363→    if (sx < -4 || sx > vw + 4 || sy < -4 || sy > vh + 4) continue
   364→    const brightness = (Math.sin(time * st.spd * 60 + st.b * 100) + 1) / 2
   365→    const alpha = 0.3 + brightness * 0.7
   366→    ctx.fillStyle = `rgba(255,255,255,${alpha.toFixed(2)})`
   367→    const s = st.size
   368→    if (s > 1.5 && brightness > 0.5) {
   369→      // Bright star with cross sparkle
   370→      ctx.fillRect(sx, sy, 2, 2)
   371→      ctx.fillStyle = `rgba(255,255,255,${(alpha * 0.25).toFixed(2)})`
   372→      ctx.fillRect(sx - 2, sy + 0.5, 6, 1)
   373→      ctx.fillRect(sx + 0.5, sy - 2, 1, 6)
   374→    } else {
   375→      ctx.fillRect(sx, sy, s > 1 ? 2 : 1, s > 1 ? 2 : 1)
   376→    }
   377→  }
   378→
   379→  // Atmospheric fog at bottom
   380→  const fogGrad = ctx.createLinearGradient(0, vh * 0.8, 0, vh)
   381→  fogGrad.addColorStop(0, 'rgba(10,15,40,0)')
   382→  fogGrad.addColorStop(1, 'rgba(10,15,40,0.3)')
   383→  ctx.fillStyle = fogGrad
   384→  ctx.fillRect(0, vh * 0.8, vw, vh * 0.2)
   385→}

   387→/* ═══════════════════ Ground Drawing ═══════════════════ */
   388→
   389→function drawGround(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number) {
   390→  // Grass patches with texture
   391→  for (const g of GRASS) {
   392→    const gx = (g.x - camX) | 0
   393→    const gy = (g.y - camY) | 0
   394→    if (gx < -16 || gx > vw + 16 || gy < -16 || gy > vh + 16) continue
   395→    ctx.fillStyle = GRASS_SHADES[g.s]
   396→    ctx.fillRect(gx, gy, g.w, g.h)
   397→    // Grass blade highlights
   398→    if (g.s % 2 === 0 && g.w > 5) {
   399→      ctx.fillStyle = 'rgba(0,229,255,0.025)'
   400→      ctx.fillRect(gx + 1, gy, 1, g.h)
   401→    }
   402→  }
   403→
   404→  // Paths with cobblestone texture
   405→  for (const p of PATHS) {
   406→    const px = (p.x - camX) | 0
   407→    const py = (p.y - camY) | 0
   408→    if (px > vw + p.w || px + p.w < 0 || py > vh + p.h || py + p.h < 0) continue
   409→    // Base path
   410→    ctx.fillStyle = '#2c2c54'
   411→    ctx.fillRect(px, py, p.w, p.h)
   412→    // Top edge highlight
   413→    ctx.fillStyle = '#3d3d6b'
   414→    ctx.fillRect(px, py, p.w, 2)
   415→    // Bottom edge shadow
   416→    ctx.fillStyle = 'rgba(0,0,0,0.35)'
   417→    ctx.fillRect(px, py + p.h - 2, p.w, 2)
   418→    // Cobblestone pattern
   419→    ctx.fillStyle = 'rgba(61,61,107,0.35)'
   420→    for (let cx = px + 5; cx < px + p.w - 5; cx += 10) {
   421→      for (let cy = py + 3; cy < py + p.h - 3; cy += 6) {
   422→        ctx.fillRect(cx + (cy % 2) * 5, cy, 7, 4)
   423→      }
   424→    }
   425→    // Path glow lines
   426→    ctx.fillStyle = 'rgba(0,229,255,0.05)'
   427→    ctx.fillRect(px + 3, py + 2, p.w - 6, 1)
   428→    ctx.fillRect(px + 3, py + p.h - 3, p.w - 6, 1)
   429→    // Center guide line
   430→    ctx.fillStyle = 'rgba(0,229,255,0.02)'
   431→    ctx.fillRect(px + 3, py + p.h / 2 - 0.5, p.w - 6, 1)
   432→  }
   433→}

   435→function drawFlowers(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number) {
   436→  for (const f of FLOWERS) {
   437→    const fx = (f.x - camX) | 0
   438→    const fy = (f.y - camY) | 0

  1140→    ctx.fillStyle = bubGrad
  1141→    roundRect(ctx, bubX, bubY, bw, bh, 6)
  1142→    ctx.fill()
  1143→
  1144→    // Inner border
  1145→    ctx.strokeStyle = 'rgba(255,255,255,0.1)'
  1146→    ctx.lineWidth = 1
  1147→    roundRect(ctx, bubX + 2, bubY + 2, bw - 4, bh - 4, 4)
  1148→    ctx.stroke()
  1149→
  1150→    // Text with outline (CLEAR, BOLD)
  1151→    ctx.textAlign = 'center'
  1152→    ctx.textBaseline = 'middle'
  1153→    drawStrokedText(ctx, text, nx, bubY + bh / 2 + 1, 3, 'rgba(0,0,0,0.9)', '#FFFFFF')
  1154→
  1155→    ctx.globalAlpha = 1
  1156→  }

  1161→function drawCrystal(ctx: CanvasRenderingContext2D, crystal: CrystalData, camX: number, camY: number, vw: number, vh: number, time: number) {
  1162→  if (crystal.collected) return
  1163→  const cx = (crystal.x - camX) | 0
  1164→  const cy = (crystal.y - camY) | 0
  1165→  if (cx < -20 || cx > vw + 20 || cy < -30 || cy > vh + 20) return
  1166→
  1167→  const floatY = Math.sin(time * 0.04 + crystal.x * 0.1) * 3
  1168→  const rgb = hexToRgb(crystal.color)
  1169→  const pulse = (Math.sin(time * 0.05 + crystal.y * 0.1) + 1) / 2
  1170→
  1171→  // Ground glow
  1172→  ctx.fillStyle = `rgba(${rgb.r},${rgb.g},${rgb.b},${(0.08 + pulse * 0.06).toFixed(3)})`
  1173→  ctx.beginPath()
  1174→  ctx.ellipse(cx, cy + 10, 12, 4, 0, 0, Math.PI * 2)
  1175→  ctx.fill()
  1176→
  1177→  // Crystal diamond shape
  1178→  const yOff = floatY
  1179→  ctx.fillStyle = crystal.color
  1180→  ctx.beginPath()
  1181→  ctx.moveTo(cx, cy - 10 + yOff)
  1182→  ctx.lineTo(cx + 6, cy + yOff)
  1183→  ctx.lineTo(cx, cy + 8 + yOff)
  1184→  ctx.lineTo(cx - 6, cy + yOff)
  1185→  ctx.closePath()
  1186→  ctx.fill()
  1187→
  1188→  // Crystal highlight
  1189→  ctx.fillStyle = `rgba(255,255,255,${(0.3 + pulse * 0.2).toFixed(2)})`
  1190→  ctx.beginPath()
  1191→  ctx.moveTo(cx - 1, cy - 8 + yOff)
  1192→  ctx.lineTo(cx + 3, cy - 1 + yOff)
  1193→  ctx.lineTo(cx, cy + 1 + yOff)
  1194→  ctx.lineTo(cx - 3, cy - 2 + yOff)
  1195→  ctx.closePath()
  1196→  ctx.fill()
  1197→
  1198→  // Crystal inner glow
  1199→  ctx.fillStyle = `rgba(${rgb.r},${rgb.g},${rgb.b},${(0.3 + pulse * 0.3).toFixed(2)})`
  1200→  ctx.fillRect(cx - 2, cy - 4 + yOff, 4, 6)
  1201→
  1202→  // Outer glow ring
  1203→  ctx.save()
  1204→  ctx.shadowColor = crystal.color
  1205→  ctx.shadowBlur = 12 + pulse * 6
  1206→  ctx.strokeStyle = `rgba(${rgb.r},${rgb.g},${rgb.b},${(0.25 + pulse * 0.2).toFixed(2)})`
  1207→  ctx.lineWidth = 1
  1208→  ctx.beginPath()
  1209→  ctx.arc(cx, cy + yOff, 10, 0, Math.PI * 2)
  1210→  ctx.stroke()
  1211→  ctx.restore()
  1212→
  1213→  // Sparkle particles
  1214→  if (time % 20 < 10) {
  1215→    ctx.fillStyle = `rgba(255,255,255,${(0.5 + pulse * 0.3).toFixed(2)})`
  1216→    ctx.fillRect(cx + 4, cy - 6 + yOff, 1, 1)
  1217→    ctx.fillRect(cx - 5, cy + 3 + yOff, 1, 1)
  1218→  }
  1219→
  1220→  // Label
  1221→  ctx.font = 'bold 6px monospace'
  1222→  ctx.textAlign = 'center'
  1223→  ctx.textBaseline = 'bottom'
  1224→  ctx.fillStyle = `rgba(${rgb.r},${rgb.g},${rgb.b},${(0.6 + pulse * 0.3).toFixed(2)})`
  1225→  ctx.fillText(crystal.label, cx, cy - 13 + yOff)
  1226→}

  1228→function drawBoostPad(ctx: CanvasRenderingContext2D, pad: BoostPadData, camX: number, camY: number, vw: number, vh: number, time: number) {
  1229→  const px = (pad.x - camX) | 0

  1231→  if (px < -24 || px > vw + 24 || py < -24 || py > vh + 24) return

  1234→  const pulse = (Math.sin(time * 0.06) + 1) / 2
