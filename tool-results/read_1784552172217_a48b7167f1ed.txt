    82→/* Seeded random for deterministic world generation */
    83→function makeRng(seed: number) {
    84→  let s = seed
    85→  return () => { s = (s * 16807) % 2147483647; return s / 2147483647 }
    86→}

   185→function hexToRgb(hex: string) {
   186→  const r = parseInt(hex.slice(1, 3), 16)
   187→  const g = parseInt(hex.slice(3, 5), 16)
   188→  const b = parseInt(hex.slice(5, 7), 16)
   189→  return { r, g, b }
   190→}
   191→
   192→function darken(hex: string, factor: number): string {
   193→  const { r, g, b } = hexToRgb(hex)
   194→  const dr = Math.round(r * factor)
   195→  const dg = Math.round(g * factor)
   196→  const db = Math.round(b * factor)
   197→  return `rgb(${dr},${dg},${db})`
   198→}
   199→
   200→function lighten(hex: string, factor: number): string {
   201→  const { r, g, b } = hexToRgb(hex)
   202→  const lr = Math.min(255, Math.round(r + (255 - r) * factor))
   203→  const lg = Math.min(255, Math.round(g + (255 - g) * factor))
   204→  const lb = Math.min(255, Math.round(b + (255 - b) * factor))
   205→  return `rgb(${lr},${lg},${lb})`
   206→}

   210→function drawBackground(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number, time: number) {
   211→  // Deep space gradient
   212→  const grad = ctx.createLinearGradient(0, 0, 0, vh)
   213→  grad.addColorStop(0, '#050816')
   214→  grad.addColorStop(0.4, '#0a0f28')
   215→  grad.addColorStop(0.7, '#0d1233')
   216→  grad.addColorStop(1, '#111640')
   217→  ctx.fillStyle = grad
   218→  ctx.fillRect(0, 0, vw, vh)
   219→
   220→  // Nebula clouds (parallax 0.15x)
   221→  for (const cloud of NEBULA_CLOUDS) {
   222→    const cx = (cloud.x * 0.15 - camX * 0.15) | 0
   223→    const cy = (cloud.y * 0.15 - camY * 0.15) | 0
   224→    if (cx + cloud.rx < -20 || cx - cloud.rx > vw + 20 || cy + cloud.ry < -20 || cy - cloud.ry > vh + 20) continue
   225→    const rgb = hexToRgb(cloud.color)
   226→    const alpha = cloud.alpha + Math.sin(time * 0.001 + cloud.x * 0.01) * 0.005
   227→    const cGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cloud.rx)
   228→    cGrad.addColorStop(0, `rgba(${rgb.r},${rgb.g},${rgb.b},${Math.max(0, alpha + 0.02).toFixed(3)})`)
   229→    cGrad.addColorStop(0.5, `rgba(${rgb.r},${rgb.g},${rgb.b},${alpha.toFixed(3)})`)
   230→    cGrad.addColorStop(1, `rgba(${rgb.r},${rgb.g},${rgb.b},0)`)
   231→    ctx.fillStyle = cGrad
   232→    ctx.beginPath()
   233→    ctx.ellipse(cx, cy, cloud.rx, cloud.ry, 0, 0, Math.PI * 2)
   234→    ctx.fill()
   235→  }
   236→
   237→  // Distant mountains (parallax 0.25x)
   238→  for (const mt of MOUNTAINS) {
   239→    const mx = (mt.x * 0.25 - camX * 0.25) | 0
   240→    const my = (vh * 0.65 - camY * 0.25) | 0
   241→    if (mx + mt.w < -10 || mx > vw + 10) continue
   242→    ctx.fillStyle = mt.color
   243→    ctx.beginPath()
   244→    ctx.moveTo(mx, my)
   245→    ctx.lineTo(mx + mt.w * 0.5, my - mt.h)
   246→    ctx.lineTo(mx + mt.w, my)
   247→    ctx.closePath()
   248→    ctx.fill()
   249→    // Mountain highlight
   250→    ctx.fillStyle = 'rgba(0,229,255,0.02)'
   251→    ctx.beginPath()
   252→    ctx.moveTo(mx + mt.w * 0.3, my - mt.h * 0.3)
   253→    ctx.lineTo(mx + mt.w * 0.5, my - mt.h)
   254→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.4)
   255→    ctx.closePath()
   256→    ctx.fill()
   257→  }
   258→
   259→  // Stars (parallax 0.4x)
   260→  for (const st of STARS) {
   261→    const sx = (st.x * 0.4 - camX * 0.4) | 0
   262→    const sy = (st.y * 0.4 - camY * 0.4) | 0
   263→    if (sx < -3 || sx > vw + 3 || sy < -3 || sy > vh + 3) continue
   264→    const brightness = (Math.sin(time * st.spd * 60 + st.b * 100) + 1) / 2
   265→    const alpha = 0.3 + brightness * 0.7
   266→    ctx.fillStyle = `rgba(255,255,255,${alpha.toFixed(2)})`
   267→    const s = st.size
   268→    if (s > 2 && brightness > 0.6) {
   269→      // Bright star with cross glow
   270→      ctx.fillRect(sx, sy, 1, 1)
   271→      ctx.fillStyle = `rgba(255,255,255,${(alpha * 0.3).toFixed(2)})`
   272→      ctx.fillRect(sx - 1, sy, 3, 1)
   273→      ctx.fillRect(sx, sy - 1, 1, 3)
   274→    } else {
   275→      ctx.fillRect(sx, sy, s > 1 ? 2 : 1, s > 1 ? 2 : 1)
   276→    }
   277→  }
   278→}

   280→function drawGround(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number) {
   281→  // Grass patches with texture
   282→  for (const g of GRASS) {
   283→    const gx = (g.x - camX) | 0
   284→    const gy = (g.y - camY) | 0
   285→    if (gx < -14 || gx > vw + 14 || gy < -14 || gy > vh + 14) continue
   286→    ctx.fillStyle = GRASS_SHADES[g.s]
   287→    ctx.fillRect(gx, gy, g.w, g.h)
   288→    // Grass blade highlights
   289→    if (g.s % 2 === 0 && g.w > 4) {
   290→      ctx.fillStyle = 'rgba(0,229,255,0.03)'
   291→      ctx.fillRect(gx + 1, gy, 1, g.h)
   292→    }
   293→  }
   294→
   295→  // Paths with cobblestone texture
   296→  for (const p of PATHS) {
   297→    const px = (p.x - camX) | 0
   298→    const py = (p.y - camY) | 0
   299→    if (px > vw + p.w || px + p.w < 0 || py > vh + p.h || py + p.h < 0) continue
   300→    // Base path
   301→    ctx.fillStyle = '#2c2c54'
   302→    ctx.fillRect(px, py, p.w, p.h)
   303→    // Top edge highlight
   304→    ctx.fillStyle = '#3d3d6b'
   305→    ctx.fillRect(px, py, p.w, 1)
   306→    // Bottom edge shadow
   307→    ctx.fillStyle = 'rgba(0,0,0,0.3)'
   308→    ctx.fillRect(px, py + p.h - 1, p.w, 1)
   309→    // Cobblestone pattern
   310→    ctx.fillStyle = 'rgba(61,61,107,0.4)'
   311→    for (let cx = px + 4; cx < px + p.w - 4; cx += 8) {
   312→      for (let cy = py + 3; cy < py + p.h - 3; cy += 5) {
   313→        ctx.fillRect(cx + (cy % 2) * 4, cy, 5, 3)
   314→      }
   315→    }
   316→    // Path glow lines
   317→    ctx.fillStyle = 'rgba(0,229,255,0.04)'
   318→    ctx.fillRect(px + 2, py + 2, p.w - 4, 1)
   319→  }
   320→}

   383→  ctx.fillRect(lx - 1, ly - 16, 4, 2)

   386→  const flicker = 0.12 + Math.sin(time * 0.008 + lamp.x) * 0.04

   388→  // Downward cone
   389→  ctx.beginPath()

   391→  ctx.lineTo(lx - 10, ly + 4)
   392→  ctx.lineTo(lx + 12, ly + 4)
   393→  ctx.lineTo(lx + 4, ly - 14)
   394→  ctx.closePath()
   395→  ctx.fill()
   396→
   397→  // Soft circular glow
   398→  const glowGrad = ctx.createRadialGradient(lx + 1, ly - 14, 0, lx + 1, ly - 14, 18)
   399→  glowGrad.addColorStop(0, `rgba(255,249,196,${(flicker * 0.8).toFixed(3)})`)
   400→  glowGrad.addColorStop(1, 'rgba(255,249,196,0)')
   401→  ctx.fillStyle = glowGrad
   402→  ctx.fillRect(lx - 18, ly - 32, 38, 38)
   403→}

   871→    ctx.fill()
   872→
   873→    // Tail pointing down
   874→    ctx.beginPath()
   875→    ctx.moveTo(nx - 3, bubY + bh)
   876→    ctx.lineTo(nx, bubY + bh + 6)
   877→    ctx.lineTo(nx + 3, bubY + bh)
   878→    ctx.closePath()
   879→    ctx.fill()
   880→
   881→    // Border with glow
   882→    ctx.strokeStyle = npc.color
   883→    ctx.lineWidth = 1
   884→    ctx.shadowColor = npc.color
   885→    ctx.shadowBlur = 4
   886→    roundRect(ctx, bubX, bubY, bw, bh, 4)
   887→    ctx.stroke()
   888→    ctx.shadowBlur = 0
   889→
   890→    // Top border puffs
   891→    ctx.beginPath()
   892→    ctx.arc(bubX + 6, bubY, 4, 0, Math.PI * 2)
   893→    ctx.arc(bubX + bw / 2, bubY - 2, 5, 0, Math.PI * 2)
   894→    ctx.arc(bubX + bw - 6, bubY, 4, 0, Math.PI * 2)
   895→    ctx.stroke()
   896→
   897→    // Gradient overlay for depth
   898→    const bubGrad = ctx.createLinearGradient(bubX, bubY, bubX, bubY + bh)
   899→    bubGrad.addColorStop(0, 'rgba(0,229,255,0.06)')
   900→    bubGrad.addColorStop(1, 'rgba(108,99,255,0.04)')
   901→    ctx.fillStyle = bubGrad
   902→    roundRect(ctx, bubX, bubY, bw, bh, 4)
   903→    ctx.fill()
   904→
   905→    // Text with outline for clarity
   906→    ctx.textAlign = 'center'
   907→    ctx.textBaseline = 'middle'
   908→    // Text outline
   909→    ctx.fillStyle = '#000000'
   910→    ctx.fillText(text, nx + 0.4, bubY + bh / 2 + 0.4)
   911→    ctx.fillText(text, nx - 0.4, bubY + bh / 2 - 0.4)
   912→    // Main text
   913→    ctx.fillStyle = '#FFFFFF'
   914→    ctx.fillText(text, nx, bubY + bh / 2)
   915→
   916→    ctx.globalAlpha = 1
   917→  }
   918→}

   920→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   921→  ctx.beginPath()
   922→  ctx.moveTo(x + r, y)
   923→  ctx.lineTo(x + w - r, y)
   924→  ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   925→  ctx.lineTo(x + w, y + h - r)
   926→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   927→  ctx.lineTo(x + r, y + h)
   928→  ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   929→  ctx.lineTo(x, y + r)
   930→  ctx.quadraticCurveTo(x, y, x + r, y)
   931→  ctx.closePath()
   932→}

   936→function drawCharacter(
   937→  ctx: CanvasRenderingContext2D,
   938→  screenX: number,
   939→  screenY: number,
   940→  dir: string,
   941→  frame: number,
   942→  isMoving: boolean,
   943→  idleBob: number,
   944→  time: number
   945→) {
   946→  const bob = isMoving ? 0 : Math.round(Math.sin(idleBob * 0.12) * 1.5)
   947→  const walkBob = isMoving ? Math.round(Math.sin(time * 0.018) * 1) : 0
   948→
   949→  ctx.save()
   950→  ctx.translate(screenX | 0, (screenY + bob) | 0)
   951→
   952→  // Shadow
   953→  ctx.fillStyle = 'rgba(0,0,0,0.25)'
   954→  ctx.beginPath()

   961→    ctx.fillRect(3, 16, 4, 8)
   962→    ctx.fillRect(9, 16, 4, 8)
   963→  } else if (frame === 1) {
   964→    ctx.fillRect(2, 16, 4, 8)
   965→    ctx.fillRect(10, 18, 4, 6)
   966→  } else if (frame === 3) {
   967→    ctx.fillRect(4, 18, 4, 6)
   968→    ctx.fillRect(8, 16, 4, 8)
   969→  } else {
   970→    ctx.fillRect(3, 17, 4, 7)
   971→    ctx.fillRect(9, 17, 4, 7)
   972→  }
   973→  // Shoes
   974→  ctx.fillStyle = '#004D40'
   975→  if (!isMoving) {
   976→    ctx.fillRect(2, 22, 5, 2)
   977→    ctx.fillRect(9, 22, 5, 2)
   978→  }
   979→
   980→  // Body
   981→  ctx.fillStyle = '#00E5FF'
   982→  ctx.fillRect(2, 8 + walkBob, 12, 8)
   983→  // Body stripe
   984→  ctx.fillStyle = '#00BCD4'
   985→  ctx.fillRect(2, 12 + walkBob, 12, 1)
   986→  // Body highlight
   987→  ctx.fillStyle = 'rgba(255,255,255,0.1)'
   988→  ctx.fillRect(2, 8 + walkBob, 12, 1)
   989→  // Pocket
   990→  ctx.fillStyle = '#00838F'
   991→  ctx.fillRect(4, 10 + walkBob, 3, 2)
   992→
   993→  // Arms
   994→  const armSwing = isMoving ? (frame === 1 ? -2 : frame === 3 ? 2 : 0) : 0
   995→  ctx.fillStyle = '#00BCD4'
   996→  ctx.fillRect(-1, 9 + walkBob + armSwing, 3, 6)
   997→  ctx.fillRect(14, 9 + walkBob - armSwing, 3, 6)
   998→  // Hands
   999→  ctx.fillStyle = '#FFE0B2'
  1000→  ctx.fillRect(-1, 14 + walkBob + armSwing, 3, 2)
  1001→  ctx.fillRect(14, 14 + walkBob - armSwing, 3, 2)
  1002→
  1003→  // Head
  1004→  ctx.fillStyle = '#FFE0B2'
  1005→  ctx.fillRect(3, 0, 10, 8)
  1006→
  1007→  // Hair
  1008→  ctx.fillStyle = '#4E342E'
  1009→  ctx.fillRect(3, 0, 10, 3)
  1010→  if (dir === 'left') ctx.fillRect(3, 0, 2, 5)
  1011→  else if (dir === 'right') ctx.fillRect(11, 0, 2, 5)
  1012→
  1013→  // Eyes
  1014→  ctx.fillStyle = '#212121'
  1015→  if (dir === 'down') {
  1016→    ctx.fillRect(5, 4, 2, 2)
  1017→    ctx.fillRect(9, 4, 2, 2)
  1018→    ctx.fillRect(7, 6, 2, 1)
  1019→  } else if (dir === 'up') {
  1020→    ctx.fillStyle = '#4E342E'
  1021→    ctx.fillRect(3, 2, 10, 2)
  1022→  } else if (dir === 'left') {
  1023→    ctx.fillRect(4, 4, 2, 2)
  1024→    ctx.fillRect(7, 4, 2, 2)
  1025→    ctx.fillRect(5, 6, 2, 1)
  1026→  } else {
  1027→    ctx.fillRect(7, 4, 2, 2)
  1028→    ctx.fillRect(10, 4, 2, 2)
  1029→    ctx.fillRect(9, 6, 2, 1)
  1030→  }
  1031→
  1032→  ctx.restore()
  1033→}

  1037→function drawMinimap(ctx: CanvasRenderingContext2D, vw: number, vh: number, charX: number, charY: number, camX: number, camY: number) {
  1038→  const mw = 100
  1039→  const mh = 75

  1092→  ctx.textAlign = 'left'
  1093→  ctx.textBaseline = 'top'

  1097→function drawPrompt(ctx: CanvasRenderingContext2D, building: BuildingData | null, camX: number, camY: number, isMobile: boolean, time: number) {
  1098→  if (!building) return
  1099→
  1100→  const bx = (building.x + building.w / 2 - camX) | 0
  1101→  const by = (building.y + building.h + 16 - camY) | 0
  1102→  const text = isMobile ? 'Tap [E] to enter' : 'Press E to enter'
  1103→
  1104→  ctx.font = 'bold 7px monospace'
  1105→  const tw = ctx.measureText(text).width
  1106→  const pw = tw + 14
  1107→  const ph = 15
  1108→
  1109→  // Pulsing alpha
  1110→  const alpha = 0.7 + Math.sin(time * 0.006) * 0.3
  1111→  ctx.globalAlpha = alpha
  1112→
  1113→  // Background pill
  1114→  ctx.fillStyle = 'rgba(0,0,0,0.8)'
  1115→  roundRect(ctx, bx - pw / 2, by, pw, ph, 4)
  1116→  ctx.fill()

  1118→  // Border with building color
  1119→  ctx.strokeStyle = building.glowColor
  1120→  ctx.lineWidth = 1
  1121→  ctx.shadowColor = building.glowColor
  1122→  ctx.shadowBlur = 4
  1123→  roundRect(ctx, bx - pw / 2, by, pw, ph, 4)
  1124→  ctx.stroke()
  1125→  ctx.shadowBlur = 0
  1126→
  1127→  // Gradient fill
  1128→  const pGrad = ctx.createLinearGradient(bx - pw / 2, by, bx - pw / 2, by + ph)
  1129→  pGrad.addColorStop(0, 'rgba(0,229,255,0.08)')

  1148→export default function PixelWorld({ onBuildingClick }: PixelWorldProps) {

  1172→  const thumbRef = useRef<HTMLDivElement>(null)

  1177→  const [showHelp, setShowHelp] = useState(true)

  1181→    const check = () => setIsMobile('ontouchstart' in window || navigator.maxTouchPoints > 0)
  1182→    check()
  1183→    const onResize = () => check()
  1184→    window.addEventListener('resize', onResize)
  1185→    return () => window.removeEventListener('resize', onResize)
  1186→  }, [])
  1187→
  1188→  // Hide help
  1189→  useEffect(() => {
  1190→    const t = setTimeout(() => setShowHelp(false), 6000)
  1191→    return () => clearTimeout(t)
  1192→  }, [])
  1193→
  1194→  // Find nearest building
  1195→  const findNearBuilding = useCallback((cx: number, cy: number): BuildingData | null => {
  1196→    let nearest: BuildingData | null = null
  1197→    let minDist = INTERACT_DIST
  1198→    for (const b of BUILDINGS) {
  1199→      const d = Math.hypot(cx - (b.x + b.w / 2), cy - (b.y + b.h / 2))
  1200→      if (d < minDist) { minDist = d; nearest = b }
  1201→    }
  1202→    return nearest
  1203→  }, [])
  1204→

  1207→    const s = stateRef.current
  1208→    const b = findNearBuilding(s.charX + 8, s.charY + 12)
  1209→    if (b) onBuildingClick(b.id)
  1210→  }, [onBuildingClick, findNearBuilding])

  1215→      const k = e.key.toLowerCase()
  1216→      keysRef.current.add(k)
  1217→      if (k === 'e' || k === 'enter') interact()

  1220→    window.addEventListener('keydown', onDown)
  1221→    window.addEventListener('keyup', onUp)
  1222→    return () => { window.removeEventListener('keydown', onDown); window.removeEventListener('keyup', onUp) }
  1223→  }, [interact])
  1224→
  1225→  // Canvas click-to-move
  1226→  const handleCanvasClick = useCallback((e: React.MouseEvent<HTMLCanvasElement>) => {
  1227→    const canvas = canvasRef.current
  1228→    if (!canvas) return
  1229→    const rect = canvas.getBoundingClientRect()
  1230→    const s = stateRef.current
  1231→    const offscreen = offscreenRef.current
  1232→    if (!offscreen) return
  1233→    const scaleX = offscreen.width / rect.width
  1234→    const scaleY = offscreen.height / rect.height
  1235→    s.targetX = (e.clientX - rect.left) * scaleX + s.camX
  1236→    s.targetY = (e.clientY - rect.top) * scaleY + s.camY
  1237→  }, [])
  1238→
  1239→  // Touch tap-to-move on canvas
  1240→  const handleCanvasTouch = useCallback((e: React.TouchEvent<HTMLCanvasElement>) => {
  1241→    const canvas = canvasRef.current
  1242→    if (!canvas) return
  1243→    const rect = canvas.getBoundingClientRect()
  1244→    const touch = e.changedTouches[0]
  1245→    if (touch.clientY > rect.bottom - 130) return
  1246→    const s = stateRef.current
  1247→    const offscreen = offscreenRef.current
  1248→    if (!offscreen) return
  1249→    const scaleX = offscreen.width / rect.width
  1250→    const scaleY = offscreen.height / rect.height
  1251→    s.targetX = (touch.clientX - rect.left) * scaleX + s.camX
  1252→    s.targetY = (touch.clientY - rect.top) * scaleY + s.camY
  1253→  }, [])
  1254→
  1255→  // Joystick handlers
  1256→  const handleJoyStart = useCallback((e: React.TouchEvent) => {
  1257→    e.preventDefault(); e.stopPropagation()
  1258→    joystickRef.current.active = true
  1259→    joystickRef.current.dx = 0; joystickRef.current.dy = 0
  1260→    stateRef.current.targetX = -1; stateRef.current.targetY = -1
  1261→    if (thumbRef.current) thumbRef.current.style.transform = 'translate(0px, 0px)'
  1262→  }, [])
  1263→
  1264→  const handleJoyMove = useCallback((e: React.TouchEvent) => {
  1265→    e.preventDefault(); e.stopPropagation()
  1266→    if (!joystickRef.current.active) return
  1267→    const touch = e.touches[0]
  1268→    const base = document.getElementById('joystick-base')
  1269→    if (!base) return
  1270→    const rect = base.getBoundingClientRect()
  1271→    const cx = rect.left + rect.width / 2
  1272→    const cy = rect.top + rect.height / 2
  1273→    let dx = touch.clientX - cx
  1274→    let dy = touch.clientY - cy
  1275→    const maxR = rect.width / 2 - 20
  1276→    const dist = Math.hypot(dx, dy)
  1277→    if (dist > maxR) { dx = dx / dist * maxR; dy = dy / dist * maxR }
  1278→    joystickRef.current.dx = dx / maxR
  1279→    joystickRef.current.dy = dy / maxR
  1280→    if (thumbRef.current) thumbRef.current.style.transform = `translate(${dx}px, ${dy}px)`
  1281→  }, [])
  1282→
  1283→  const handleJoyEnd = useCallback((e: React.TouchEvent) => {
  1284→    e.preventDefault()
  1285→    joystickRef.current.active = false
  1286→    joystickRef.current.dx = 0; joystickRef.current.dy = 0
  1287→    if (thumbRef.current) thumbRef.current.style.transform = 'translate(0px, 0px)'
  1288→  }, [])
  1289→
  1290→  // ── Main Game Loop Effect ──
  1291→  useEffect(() => {
  1292→    const offscreen = document.createElement('canvas')
  1293→    offscreenRef.current = offscreen
  1294→
  1295→    const resize = () => {
  1296→      const canvas = canvasRef.current
  1297→      const container = containerRef.current
  1298→      if (!canvas || !container) return
  1299→      const cw = container.clientWidth
  1300→      const ch = container.clientHeight
  1301→      canvas.width = cw
  1302→      canvas.height = ch
  1303→      offscreen.width = Math.ceil(cw / PIXEL_SCALE)
  1304→      offscreen.height = Math.ceil(ch / PIXEL_SCALE)
  1305→    }
  1306→    resize()
  1307→    window.addEventListener('resize', resize)
  1308→
  1309→    const loop = () => {
  1310→      const canvas = canvasRef.current
  1311→      const ctx = canvas?.getContext('2d')
  1312→      const offCtx = offscreen?.getContext('2d')
  1313→      if (!canvas || !ctx || !offCtx) { animRef.current = requestAnimationFrame(loop); return }
  1314→
  1315→      const s = stateRef.current
  1316→      const keys = keysRef.current
  1317→      const joy = joystickRef.current
  1318→      timeRef.current++
  1319→      const time = timeRef.current
  1320→      const vw = offscreen.width
  1321→      const vh = offscreen.height
  1322→
  1323→      // ── Input Processing ──
  1324→      let dx = 0, dy = 0
  1325→      if (keys.has('arrowleft') || keys.has('a')) dx -= 1
  1326→      if (keys.has('arrowright') || keys.has('d')) dx += 1
  1327→      if (keys.has('arrowup') || keys.has('w')) dy -= 1
  1328→      if (keys.has('arrowdown') || keys.has('s')) dy += 1
  1329→
  1330→      if (joy.active && (Math.abs(joy.dx) > 0.15 || Math.abs(joy.dy) > 0.15)) {
  1331→        dx += joy.dx
  1332→        dy += joy.dy
  1333→      }
  1334→
  1335→      // Click/tap to move
  1336→      if (s.targetX >= 0 && s.targetY >= 0 && !joy.active && dx === 0 && dy === 0) {
  1337→        const tdx = s.targetX - (s.charX + 8)
  1338→        const tdy = s.targetY - (s.charY + 12)
  1339→        const tdist = Math.hypot(tdx, tdy)
  1340→        if (tdist > CHAR_SPEED + 1) {
  1341→          dx += tdx / tdist
  1342→          dy += tdy / tdist
  1343→        } else {
  1344→          s.targetX = -1; s.targetY = -1
  1345→        }
  1346→      } else if (dx !== 0 || dy !== 0) {
  1347→        s.targetX = -1; s.targetY = -1
  1348→      }
  1349→
  1350→      // Normalize
  1351→      const len = Math.hypot(dx, dy)
  1352→      if (len > 0.01) {
  1353→        const speed = CHAR_SPEED
  1354→        dx = dx / len * speed
  1355→        dy = dy / len * speed
  1356→        s.isMoving = true
  1357→
  1358→        if (Math.abs(dx) > Math.abs(dy)) s.dir = dx > 0 ? 'right' : 'left'
  1359→        else s.dir = dy > 0 ? 'down' : 'up'
  1360→
