   362→   362→function drawNPC(ctx: CanvasRenderingContext2D, npc: NPCData, camX: number, camY: number, vw: number, vh: number, charX: number, charY: number, time: number) {

   366→   366→
   367→   367→  const dist = Math.hypot(npc.x - charX - 8, npc.y - charY - 12)
   368→   368→
   369→   369→  if (npc.type === 'robot') {
   370→   370→    // Body

   372→   372→    ctx.fillRect(nx - 5, ny - 12, 10, 12)
   373→   373→    // Head

   375→   375→    ctx.fillRect(nx - 4, ny - 20, 8, 8)
   376→   376→    // Eyes

   378→   378→    ctx.fillStyle = '#F44336'
   379→   379→    ctx.fillRect(nx - 2, ny - 17, 2, 2 * blink)
   380→   380→    ctx.fillRect(nx + 1, ny - 17, 2, 2 * blink)
   381→   381→    // Antenna

   383→   383→    ctx.fillRect(nx, ny - 23, 1, 3)
   384→   384→    ctx.fillRect(nx - 1, ny - 25, 3, 2)
   385→   385→    // Hover bob
   386→   386→    const hb = Math.sin(time * 0.006 + npc.x) * 1.5
   387→   387→    if (Math.abs(hb) > 0.5) {
   388→   388→      ctx.fillStyle = 'rgba(0,0,0,0.15)'
   389→   389→      ctx.fillRect(nx - 4, ny + 1, 8, 2)
   390→   390→    }
   391→   391→  } else {
   392→   392→    // Lab coat body
   393→   393→    ctx.fillStyle = '#ECEFF1'
   394→   394→    ctx.fillRect(nx - 5, ny - 12, 10, 14)
   395→   395→    // Head
   396→   396→    ctx.fillStyle = '#FFCCBC'
   397→   397→    ctx.fillRect(nx - 4, ny - 20, 8, 8)
   398→   398→    // Hair
   399→   399→    ctx.fillStyle = '#5D4037'
   400→   400→    ctx.fillRect(nx - 4, ny - 22, 8, 4)
   401→   401→    // Glasses
   402→   402→    ctx.fillStyle = '#263238'
   403→   403→    ctx.fillRect(nx - 3, ny - 17, 3, 2)
   404→   404→    ctx.fillRect(nx + 1, ny - 17, 3, 2)
   405→   405→    ctx.fillRect(nx, ny - 17, 1, 2)
   406→   406→    // Legs
   407→   407→    ctx.fillStyle = '#455A64'
   408→   408→    ctx.fillRect(nx - 3, ny + 2, 3, 4)
   409→   409→    ctx.fillRect(nx + 1, ny + 2, 3, 4)
   410→   410→  }
   411→   411→
   412→   412→  // Speech bubble
   413→   413→  if (dist < NPC_DIST) {
   414→   414→    const alpha = Math.min(1, (NPC_DIST - dist) / 20)
   415→   415→    ctx.globalAlpha = alpha
   416→   416→    const text = npc.message
   417→   417→    ctx.font = '5px monospace'
   418→   418→    const tw = ctx.measureText(text).width
   419→   419→    const bw = tw + 8
   420→   420→    const bh = 11
   421→   421→    const bubX = (nx - bw / 2) | 0
   422→   422→    const bubY = ny - (npc.type === 'robot' ? 38 : 38)
   423→   423→
   424→   424→    ctx.fillStyle = '#FFFFFF'
   425→   425→    ctx.fillRect(bubX, bubY, bw, bh)
   426→   426→    // Tail
   427→   427→    ctx.fillRect(nx - 1, bubY + bh, 2, 3)
   428→   428→    // Border
   429→   429→    ctx.strokeStyle = '#000000'
   430→   430→    ctx.lineWidth = 0.5
   431→   431→    ctx.strokeRect(bubX, bubY, bw, bh)
   432→   432→    // Text
   433→   433→    ctx.fillStyle = '#212121'
   434→   434→    ctx.textAlign = 'center'
   435→   435→    ctx.textBaseline = 'middle'
   436→   436→    ctx.fillText(text, nx, bubY + bh / 2 + 1)
   437→   437→    ctx.globalAlpha = 1
   438→   438→  }
   439→   439→}

   441→   441→function drawCharacter(
   442→   442→  ctx: CanvasRenderingContext2D,
   443→   443→  screenX: number,
   444→   444→  screenY: number,

   450→   450→) {

   502→   502→  if (dir === 'down') {
   503→   503→    ctx.fillRect(5, 4, 2, 2)
   504→   504→    ctx.fillRect(9, 4, 2, 2)

   520→   520→}

   522→   522→function drawMinimap(ctx: CanvasRenderingContext2D, vw: number, vh: number, charX: number, charY: number, camX: number, camY: number) {

   562→   562→function drawPrompt(ctx: CanvasRenderingContext2D, building: BuildingData | null, camX: number, camY: number, isMobile: boolean, time: number) {
   563→   563→  if (!building) return
   564→   564→
   565→   565→  const bx = (building.x + building.w / 2 - camX) | 0

   568→   568→
   569→   569→  ctx.font = '6px monospace'

   571→   571→  const pw = tw + 10
   572→   572→  const ph = 13
   573→   573→

   578→   578→  ctx.fillStyle = 'rgba(0,0,0,0.75)'

   581→   581→  ctx.fillStyle = building.glowColor
   582→   582→  ctx.textAlign = 'center'
   583→   583→  ctx.textBaseline = 'middle'
   584→   584→  ctx.fillText(text, bx, by + ph / 2)
   585→   585→
   586→   586→  ctx.globalAlpha = 1
   587→   587→}

   591→   591→export default function PixelWorld({ onBuildingClick }: PixelWorldProps) {
   592→   592→  const canvasRef = useRef<HTMLCanvasElement>(null)
   593→   593→  const containerRef = useRef<HTMLDivElement>(null)
   594→   594→  const offscreenRef = useRef<HTMLCanvasElement | null>(null)
   595→   595→  const animRef = useRef(0)
   596→   596→  const timeRef = useRef(0)
   597→   597→
   598→   598→  const stateRef = useRef({
   599→   599→    charX: WORLD_W / 2 - 8,
   600→   600→    charY: WORLD_H / 2 - 12,
   601→   601→    dir: 'down' as 'left' | 'right' | 'up' | 'down',
   602→   602→    frame: 0,
   603→   603→    frameTimer: 0,
   604→   604→    isMoving: false,
   605→   605→    camX: 0,
   606→   606→    camY: 0,
   607→   607→    nearBuilding: null as string | null,
   608→   608→    idleBob: 0,
   609→   609→    targetX: -1,
   610→   610→    targetY: -1,
   611→   611→  })
   612→   612→
   613→   613→  const keysRef = useRef(new Set<string>())
   614→   614→  const joystickRef = useRef({ active: false, dx: 0, dy: 0 })
   615→   615→  const thumbRef = useRef<HTMLDivElement>(null)
   616→   616→  const particlesRef = useRef<Particle[]>([])
   617→   617→
   618→   618→  const [isMobile, setIsMobile] = useState(false)
   619→   619→  const [nearBuilding, setNearBuilding] = useState<string | null>(null)
   620→   620→  const [showHelp, setShowHelp] = useState(true)
   621→   621→
   622→   622→  // Detect mobile
   623→   623→  useEffect(() => {
   624→   624→    const check = () => setIsMobile('ontouchstart' in window || navigator.maxTouchPoints > 0)
   625→   625→    check()
   626→   626→    const onResize = () => check()
   627→   627→    window.addEventListener('resize', onResize)
   628→   628→    return () => window.removeEventListener('resize', onResize)
   629→   629→  }, [])
   630→   630→
   631→   631→  // Hide help
   632→   632→  useEffect(() => {
   633→   633→    const t = setTimeout(() => setShowHelp(false), 6000)
   634→   634→    return () => clearTimeout(t)
   635→   635→  }, [])
   636→   636→
   637→   637→  // Find nearest building
   638→   638→  const findNearBuilding = useCallback((cx: number, cy: number): BuildingData | null => {
   639→   639→    let nearest: BuildingData | null = null
   640→   640→    let minDist = INTERACT_DIST
   641→   641→    for (const b of BUILDINGS) {
   642→   642→      const d = Math.hypot(cx - (b.x + b.w / 2), cy - (b.y + b.h / 2))
   643→   643→      if (d < minDist) { minDist = d; nearest = b }
   644→   644→    }
   645→   645→    return nearest
   646→   646→  }, [])
   647→   647→
   648→   648→  // Interact
   649→   649→  const interact = useCallback(() => {
   650→   650→    const s = stateRef.current
   651→   651→    const b = findNearBuilding(s.charX + 8, s.charY + 12)
   652→   652→    if (b) onBuildingClick(b.id)
   653→   653→  }, [onBuildingClick, findNearBuilding])
   654→   654→
   655→   655→  // Keyboard
   656→   656→  useEffect(() => {
   657→   657→    const onDown = (e: KeyboardEvent) => {
   658→   658→      const k = e.key.toLowerCase()
   659→   659→      keysRef.current.add(k)
   660→   660→      if (k === 'e' || k === 'enter') interact()
   661→   661→    }
   662→   662→    const onUp = (e: KeyboardEvent) => keysRef.current.delete(e.key.toLowerCase())
   663→   663→    window.addEventListener('keydown', onDown)
   664→   664→    window.addEventListener('keyup', onUp)
   665→   665→    return () => { window.removeEventListener('keydown', onDown); window.removeEventListener('keyup', onUp) }
   666→   666→  }, [interact])
   667→   667→
   668→   668→  // Canvas click-to-move
   669→   669→  const handleCanvasClick = useCallback((e: React.MouseEvent<HTMLCanvasElement>) => {
   670→   670→    const canvas = canvasRef.current
   671→   671→    if (!canvas) return
   672→   672→    const rect = canvas.getBoundingClientRect()
   673→   673→    const s = stateRef.current
   674→   674→    const offscreen = offscreenRef.current
   675→   675→    if (!offscreen) return
   676→   676→    const scaleX = offscreen.width / rect.width
   677→   677→    const scaleY = offscreen.height / rect.height
   678→   678→    s.targetX = (e.clientX - rect.left) * scaleX + s.camX
   679→   679→    s.targetY = (e.clientY - rect.top) * scaleY + s.camY
   680→   680→  }, [])
   681→   681→
   682→   682→  // Touch tap-to-move on canvas
   683→   683→  const handleCanvasTouch = useCallback((e: React.TouchEvent<HTMLCanvasElement>) => {
   684→   684→    const canvas = canvasRef.current
   685→   685→    if (!canvas) return
   686→   686→    const rect = canvas.getBoundingClientRect()
   687→   687→    const touch = e.changedTouches[0]
   688→   688→    if (touch.clientY > rect.bottom - 130) return
   689→   689→    const s = stateRef.current
   690→   690→    const offscreen = offscreenRef.current
   691→   691→    if (!offscreen) return
   692→   692→    const scaleX = offscreen.width / rect.width
   693→   693→    const scaleY = offscreen.height / rect.height
   694→   694→    s.targetX = (touch.clientX - rect.left) * scaleX + s.camX
   695→   695→    s.targetY = (touch.clientY - rect.top) * scaleY + s.camY
   696→   696→  }, [])
   697→   697→
   698→   698→  // Joystick handlers
   699→   699→  const handleJoyStart = useCallback((e: React.TouchEvent) => {
   700→   700→    e.preventDefault(); e.stopPropagation()
   701→   701→    joystickRef.current.active = true
   702→   702→    joystickRef.current.dx = 0; joystickRef.current.dy = 0
   703→   703→    stateRef.current.targetX = -1; stateRef.current.targetY = -1
   704→   704→    if (thumbRef.current) thumbRef.current.style.transform = 'translate(0px, 0px)'
   705→   705→  }, [])
   706→   706→
   707→   707→  const handleJoyMove = useCallback((e: React.TouchEvent) => {
   708→   708→    e.preventDefault(); e.stopPropagation()
   709→   709→    if (!joystickRef.current.active) return
   710→   710→    const touch = e.touches[0]
   711→   711→    const base = document.getElementById('joystick-base')
   712→   712→    if (!base) return
   713→   713→    const rect = base.getBoundingClientRect()
   714→   714→    const cx = rect.left + rect.width / 2
   715→   715→    const cy = rect.top + rect.height / 2
   716→   716→    let dx = touch.clientX - cx
   717→   717→    let dy = touch.clientY - cy
   718→   718→    const maxR = rect.width / 2 - 20
   719→   719→    const dist = Math.hypot(dx, dy)
   720→   720→    if (dist > maxR) { dx = dx / dist * maxR; dy = dy / dist * maxR }
   721→   721→    joystickRef.current.dx = dx / maxR
   722→   722→    joystickRef.current.dy = dy / maxR
   723→   723→    if (thumbRef.current) thumbRef.current.style.transform = `translate(${dx}px, ${dy}px)`
   724→   724→  }, [])
   725→   725→
   726→   726→  const handleJoyEnd = useCallback((e: React.TouchEvent) => {
   727→   727→    e.preventDefault()
   728→   728→    joystickRef.current.active = false
   729→   729→    joystickRef.current.dx = 0; joystickRef.current.dy = 0
   730→   730→    if (thumbRef.current) thumbRef.current.style.transform = 'translate(0px, 0px)'
   731→   731→  }, [])
   732→   732→
   733→   733→  // ── Main Game Loop Effect ──
   734→   734→  useEffect(() => {
   735→   735→    const offscreen = document.createElement('canvas')
   736→   736→    offscreenRef.current = offscreen
   737→   737→
   738→   738→    const resize = () => {
   739→   739→      const canvas = canvasRef.current
   740→   740→      const container = containerRef.current
   741→   741→      if (!canvas || !container) return
   742→   742→      const cw = container.clientWidth
   743→   743→      const ch = container.clientHeight
   744→   744→      canvas.width = cw
   745→   745→      canvas.height = ch
   746→   746→      offscreen.width = Math.ceil(cw / PIXEL_SCALE)
   747→   747→      offscreen.height = Math.ceil(ch / PIXEL_SCALE)
   748→   748→    }
   749→   749→    resize()
   750→   750→    window.addEventListener('resize', resize)
   751→   751→
   752→   752→    const loop = () => {
   753→   753→      const canvas = canvasRef.current
   754→   754→      const ctx = canvas?.getContext('2d')
   755→   755→      const offCtx = offscreen?.getContext('2d')
   756→   756→      if (!canvas || !ctx || !offCtx) { animRef.current = requestAnimationFrame(loop); return }
   757→   757→
   758→   758→      const s = stateRef.current
   759→   759→      const keys = keysRef.current
   760→   760→      const joy = joystickRef.current
   761→   761→      timeRef.current++
   762→   762→      const time = timeRef.current
   763→   763→      const vw = offscreen.width
   764→   764→      const vh = offscreen.height
   765→   765→
   766→   766→      // ── Input Processing ──
   767→   767→      let dx = 0, dy = 0
   768→   768→      if (keys.has('arrowleft') || keys.has('a')) dx -= 1
   769→   769→      if (keys.has('arrowright') || keys.has('d')) dx += 1
   770→   770→      if (keys.has('arrowup') || keys.has('w')) dy -= 1
   771→   771→      if (keys.has('arrowdown') || keys.has('s')) dy += 1
   772→   772→
   773→   773→      if (joy.active && (Math.abs(joy.dx) > 0.15 || Math.abs(joy.dy) > 0.15)) {
   774→   774→        dx += joy.dx
   775→   775→        dy += joy.dy
   776→   776→      }
   777→   777→
   778→   778→      // Click/tap to move
   779→   779→      if (s.targetX >= 0 && s.targetY >= 0 && !joy.active && dx === 0 && dy === 0) {
   780→   780→        const tdx = s.targetX - (s.charX + 8)
   781→   781→        const tdy = s.targetY - (s.charY + 12)
   782→   782→        const tdist = Math.hypot(tdx, tdy)
   783→   783→        if (tdist > CHAR_SPEED + 1) {
   784→   784→          dx += tdx / tdist
   785→   785→          dy += tdy / tdist
   786→   786→        } else {
   787→   787→          s.targetX = -1; s.targetY = -1
   788→   788→        }
   789→   789→      } else if (dx !== 0 || dy !== 0) {
   790→   790→        s.targetX = -1; s.targetY = -1
   791→   791→      }
   792→   792→
   793→   793→      // Normalize
   794→   794→      const len = Math.hypot(dx, dy)
   795→   795→      if (len > 0.01) {
   796→   796→        const speed = CHAR_SPEED
   797→   797→        dx = dx / len * speed
   798→   798→        dy = dy / len * speed
   799→   799→        s.isMoving = true
   800→   800→
   801→   801→        if (Math.abs(dx) > Math.abs(dy)) s.dir = dx > 0 ? 'right' : 'left'
   802→   802→        else s.dir = dy > 0 ? 'down' : 'up'
   803→   803→
   804→   804→        s.frameTimer++
   805→   805→        if (s.frameTimer >= 7) { s.frameTimer = 0; s.frame = (s.frame + 1) % 4 }
   806→   806→
   807→   807→        // Dust particles
   808→   808→        if (time % 8 === 0) {
   809→   809→          particlesRef.current.push({
   810→   810→            x: s.charX + 8 + (Math.random() - 0.5) * 6,
   811→   811→            y: s.charY + 23,
   812→   812→            vx: (Math.random() - 0.5) * 0.4,
   813→   813→            vy: -Math.random() * 0.3 - 0.1,
   814→   814→            life: 18, maxLife: 18,
   815→   815→            color: '#4a4a6a', size: 2,
   816→   816→          })
   817→   817→        }
   818→   818→      } else {
   819→   819→        s.isMoving = false
   820→   820→        s.frame = 0; s.frameTimer = 0
   821→   821→        s.idleBob++
   822→   822→      }
   823→   823→
   824→   824→      // Move character
   825→   825→      s.charX = Math.max(0, Math.min(WORLD_W - 16, s.charX + dx))
   826→   826→      s.charY = Math.max(0, Math.min(WORLD_H - 24, s.charY + dy))
   827→   827→
   828→   828→      // Camera (smooth follow)
   829→   829→      const targetCamX = s.charX + 8 - vw / 2
   830→   830→      const targetCamY = s.charY + 12 - vh / 2
   831→   831→      const clampedX = Math.max(0, Math.min(WORLD_W - vw, targetCamX))
   832→   832→      const clampedY = Math.max(0, Math.min(WORLD_H - vh, targetCamY))
   833→   833→      s.camX += (clampedX - s.camX) * CAM_LERP
   834→   834→      s.camY += (clampedY - s.camY) * CAM_LERP
   835→   835→
   836→   836→      // Near building detection
   837→   837→      const nearB = findNearBuilding(s.charX + 8, s.charY + 12)
   838→   838→      const newNear = nearB?.id ?? null
   839→   839→      if (newNear !== s.nearBuilding) {
   840→   840→        s.nearBuilding = newNear
   841→   841→        setNearBuilding(newNear)
   842→   842→      }
   843→   843→
   844→   844→      // Update particles
   845→   845→      particlesRef.current = particlesRef.current.filter(p => {
   846→   846→        p.x += p.vx; p.y += p.vy; p.life--
   847→   847→        return p.life > 0
   848→   848→      })
   849→   849→
   850→   850→      // ── Render ──
   851→   851→      offCtx.imageSmoothingEnabled = false
   852→   852→      const camX = s.camX
   853→   853→      const camY = s.camY
   854→   854→
   855→   855→      drawBackground(offCtx, vw, vh, camX, camY, time)
   856→   856→      drawGround(offCtx, vw, vh, camX, camY)
   857→   857→      drawFlowers(offCtx, vw, vh, camX, camY)
   858→   858→
   859→   859→      // Trees (behind buildings/character)
   860→   860→      for (const tree of TREES) drawTree(offCtx, tree, camX, camY, vw, vh)
   861→   861→
   862→   862→      // Lamps
   863→   863→      for (const lamp of LAMPS) drawLamp(offCtx, lamp, camX, camY, vw, vh, time)
   864→   864→
   865→   865→      // Center monument
   866→   866→      drawCenterMonument(offCtx, camX, camY, vw, vh, time)
   867→   867→
   868→   868→      // Y-sorted entities: buildings + NPCs + character
   869→   869→      const entities: { y: number; draw: () => void }[] = []
   870→   870→
   871→   871→      for (const b of BUILDINGS) {
   872→   872→        const bRef = b
   873→   873→        entities.push({ y: b.y + b.h, draw: () => drawBuilding(offCtx, bRef, camX, camY, vw, vh, time, s.charX, s.charY) })
   874→   874→      }
   875→   875→
   876→   876→      for (const npc of NPCS) {
   877→   877→        const nRef = npc
   878→   878→        entities.push({ y: npc.y + (npc.type === 'robot' ? 0 : 6), draw: () => drawNPC(offCtx, nRef, camX, camY, vw, vh, s.charX, s.charY, time) })
   879→   879→      }
   880→   880→
   881→   881→      entities.push({
   882→   882→        y: s.charY + 24,
   883→   883→        draw: () => drawCharacter(offCtx, s.charX - camX, s.charY - camY, s.dir, s.frame, s.isMoving, s.idleBob, time),
   884→   884→      })
   885→   885→
   886→   886→      entities.sort((a, b) => a.y - b.y)
   887→   887→      for (const e of entities) e.draw()
   888→   888→
   889→   889→      // Particles
   890→   890→      for (const p of particlesRef.current) {
   891→   891→        const px = (p.x - camX) | 0
   892→   892→        const py = (p.y - camY) | 0
   893→   893→        offCtx.globalAlpha = p.life / p.maxLife * 0.6
   894→   894→        offCtx.fillStyle = p.color
   895→   895→        offCtx.fillRect(px, py, p.size, p.size)
   896→   896→      }
   897→   897→      offCtx.globalAlpha = 1
   898→   898→
   899→   899→      // UI
   900→   900→      drawPrompt(offCtx, nearB, camX, camY, isMobile, time)
   901→   901→      drawMinimap(offCtx, vw, vh, s.charX, s.charY, camX, camY)
   902→   902→
   903→   903→      // ── Blit to display ──
   904→   904→      ctx.imageSmoothingEnabled = false
   905→   905→      ctx.clearRect(0, 0, canvas.width, canvas.height)
   906→   906→      ctx.drawImage(offscreen, 0, 0, canvas.width, canvas.height)
   907→   907→
   908→   908→      animRef.current = requestAnimationFrame(loop)
   909→   909→    }
   910→   910→
   911→   911→    animRef.current = requestAnimationFrame(loop)
   912→   912→
   913→   913→    return () => {
   914→   914→      cancelAnimationFrame(animRef.current)
   915→   915→      window.removeEventListener('resize', resize)
   916→   916→    }
   917→   917→  }, [isMobile, findNearBuilding])
   918→   918→
   919→   919→  return (
   920→   920→    <div ref={containerRef} className="relative w-full h-full overflow-hidden bg-[#1a1a2e]" style={{ minHeight: '300px' }}>
   921→   921→      <canvas
   922→   922→        ref={canvasRef}
   923→   923→        className="w-full h-full block cursor-crosshair"
   924→   924→        style={{ imageRendering: 'pixelated' }}
   925→   925→        onClick={handleCanvasClick}
   926→   926→        onTouchEnd={handleCanvasTouch}
   927→   927→      />
   928→   928→
   929→   929→      {/* Mobile virtual joystick */}
   930→   930→      {isMobile && (
   931→   931→        <div
   932→   932→          className="absolute bottom-6 left-6 touch-none select-none"
   933→   933→          onTouchStart={handleJoyStart}
   934→   934→          onTouchMove={handleJoyMove}
   935→   935→          onTouchEnd={handleJoyEnd}
   936→   936→        >
   937→   937→          <div
   938→   938→            id="joystick-base"
   939→   939→            className="w-28 h-28 rounded-full bg-white/10 border-2 border-white/20 flex items-center justify-center backdrop-blur-sm"
   940→   940→          >
   941→   941→            <div
   942→   942→              ref={thumbRef}
   943→   943→              className="w-12 h-12 rounded-full bg-white/30 border-2 border-white/40"
   944→   944→              style={{ transition: 'transform 0.05s ease-out' }}
   945→   945→            />
   946→   946→          </div>
   947→   947→        </div>
   948→   948→      )}
   949→   949→
   950→   950→      {/* Mobile action button */}
   951→   951→      {isMobile && (
   952→   952→        <button
   953→   953→          className={`absolute bottom-8 right-8 w-16 h-16 rounded-full flex items-center justify-center text-white font-bold text-xl transition-all duration-200 touch-none select-none ${
   954→   954→            nearBuilding
   955→   955→              ? 'bg-white/25 border-2 border-white/60 scale-110 shadow-lg shadow-white/20'
   956→   956→              : 'bg-white/10 border border-white/20 scale-100 opacity-40'
   957→   957→          }`}
   958→   958→          onTouchStart={(e) => { e.preventDefault(); interact() }}
   959→   959→          aria-label="Interact with building"
   960→   960→        >
   961→   961→          E
   962→   962→        </button>
   963→   963→      )}
   964→   964→
   965→   965→      {/* Help text */}
   966→   966→      {showHelp && (
   967→   967→        <div className="absolute top-4 left-1/2 -translate-x-1/2 bg-black/60 text-white/80 px-5 py-2 rounded-lg text-xs font-mono pointer-events-none backdrop-blur-sm border border-white/10 animate-pulse">
   968→   968→          {isMobile
   969→   969→            ? 'Tap to move \u2022 Joystick to walk \u2022 Tap E near buildings'
   970→   970→            : 'WASD / Arrow Keys to move \u2022 E to interact \u2022 Click to walk'}
   971→   971→        </div>
   972→   972→      )}
   973→   973→    </div>
   974→   974→  )
   975→   975→}
