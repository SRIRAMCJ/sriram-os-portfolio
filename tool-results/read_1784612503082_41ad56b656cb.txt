   186→   186→function hexToRgb(hex: string) {
   187→   187→  const r = parseInt(hex.slice(1, 3), 16)
   188→   188→  const g = parseInt(hex.slice(3, 5), 16)
   189→   189→  const b = parseInt(hex.slice(5, 7), 16)
   190→   190→  return { r, g, b }
   191→   191→}
   192→   192→
   193→   193→function darken(hex: string, factor: number): string {
   194→   194→  const { r, g, b } = hexToRgb(hex)
   195→   195→  return `rgb(${Math.round(r * factor)},${Math.round(g * factor)},${Math.round(b * factor)})`
   196→   196→}
   197→   197→
   198→   198→function lighten(hex: string, factor: number): string {
   199→   199→  const { r, g, b } = hexToRgb(hex)
   200→   200→  return `rgb(${Math.min(255, Math.round(r + (255 - r) * factor))},${Math.min(255, Math.round(g + (255 - g) * factor))},${Math.min(255, Math.round(b + (255 - b) * factor))})`
   201→   201→}

   203→   203→/* ═══════════════════ Rounded Rect Helper ═══════════════════ */
   204→   204→
   205→   205→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   206→   206→  ctx.beginPath()
   207→   207→  ctx.moveTo(x + r, y)
   208→   208→  ctx.lineTo(x + w - r, y)
   209→   209→  ctx.quadraticCurveTo(x + w, y, x + w, y + r)
   210→   210→  ctx.lineTo(x + w, y + h - r)
   211→   211→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h)
   212→   212→  ctx.lineTo(x + r, y + h)
   213→   213→  ctx.quadraticCurveTo(x, y + h, x, y + h - r)
   214→   214→  ctx.lineTo(x, y + r)
   215→   215→  ctx.quadraticCurveTo(x, y, x + r, y)
   216→   216→  ctx.closePath()
   217→   217→}

   252→   252→
   253→   253→function drawBackground(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number, time: number) {
   254→   254→  // Deep space gradient
   255→   255→  const grad = ctx.createLinearGradient(0, 0, 0, vh)
   256→   256→  grad.addColorStop(0, '#030712')
   257→   257→  grad.addColorStop(0.3, '#070d1f')
   258→   258→  grad.addColorStop(0.6, '#0a1030')
   259→   259→  grad.addColorStop(1, '#0e1545')
   260→   260→  ctx.fillStyle = grad
   261→   261→  ctx.fillRect(0, 0, vw, vh)
   262→   262→
   263→   263→  // Nebula clouds (parallax 0.12x)
   264→   264→  for (const cloud of NEBULA_CLOUDS) {
   265→   265→    const cx = (cloud.x * 0.12 - camX * 0.12) | 0
   266→   266→    const cy = (cloud.y * 0.12 - camY * 0.12) | 0
   267→   267→    if (cx + cloud.rx < -20 || cx - cloud.rx > vw + 20 || cy + cloud.ry < -20 || cy - cloud.ry > vh + 20) continue
   268→   268→    const rgb = hexToRgb(cloud.color)
   269→   269→    const alpha = cloud.alpha + Math.sin(time * 0.001 + cloud.x * 0.01) * 0.006
   270→   270→    const cGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cloud.rx)
   271→   271→    cGrad.addColorStop(0, `rgba(${rgb.r},${rgb.g},${rgb.b},${(alpha + 0.03).toFixed(3)})`)
   272→   272→    cGrad.addColorStop(0.4, `rgba(${rgb.r},${rgb.g},${rgb.b},${alpha.toFixed(3)})`)
   273→   273→    cGrad.addColorStop(1, `rgba(${rgb.r},${rgb.g},${rgb.b},0)`)
   274→   274→    ctx.fillStyle = cGrad
   275→   275→    ctx.beginPath()
   276→   276→    ctx.ellipse(cx, cy, cloud.rx, cloud.ry, 0, 0, Math.PI * 2)
   277→   277→    ctx.fill()
   278→   278→  }
   279→   279→
   280→   280→  // Distant mountains (parallax 0.2x)
   281→   281→  for (const mt of MOUNTAINS) {
   282→   282→    const mx = (mt.x * 0.2 - camX * 0.2) | 0
   283→   283→    const my = (vh * 0.6 - camY * 0.2) | 0
   284→   284→    if (mx + mt.w < -10 || mx > vw + 10) continue
   285→   285→    ctx.fillStyle = mt.color
   286→   286→    ctx.beginPath()
   287→   287→    ctx.moveTo(mx, my)
   288→   288→    ctx.lineTo(mx + mt.w * 0.45, my - mt.h)
   289→   289→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.9)
   290→   290→    ctx.lineTo(mx + mt.w, my)
   291→   291→    ctx.closePath()
   292→   292→    ctx.fill()
   293→   293→    // Mountain snow cap
   294→   294→    ctx.fillStyle = 'rgba(0,229,255,0.04)'
   295→   295→    ctx.beginPath()
   296→   296→    ctx.moveTo(mx + mt.w * 0.4, my - mt.h * 0.7)
   297→   297→    ctx.lineTo(mx + mt.w * 0.45, my - mt.h)
   298→   298→    ctx.lineTo(mx + mt.w * 0.55, my - mt.h * 0.9)
   299→   299→    ctx.lineTo(mx + mt.w * 0.6, my - mt.h * 0.6)
   300→   300→    ctx.closePath()
   301→   301→    ctx.fill()
   302→   302→  }
   303→   303→
   304→   304→  // Stars (parallax 0.35x)
   305→   305→  for (const st of STARS) {
   306→   306→    const sx = (st.x * 0.35 - camX * 0.35) | 0
   307→   307→    const sy = (st.y * 0.35 - camY * 0.35) | 0
   308→   308→    if (sx < -4 || sx > vw + 4 || sy < -4 || sy > vh + 4) continue
   309→   309→    const brightness = (Math.sin(time * st.spd * 60 + st.b * 100) + 1) / 2
   310→   310→    const alpha = 0.3 + brightness * 0.7
   311→   311→    ctx.fillStyle = `rgba(255,255,255,${alpha.toFixed(2)})`
   312→   312→    const s = st.size
   313→   313→    if (s > 1.5 && brightness > 0.5) {
   314→   314→      // Bright star with cross sparkle
   315→   315→      ctx.fillRect(sx, sy, 2, 2)
   316→   316→      ctx.fillStyle = `rgba(255,255,255,${(alpha * 0.25).toFixed(2)})`
   317→   317→      ctx.fillRect(sx - 2, sy + 0.5, 6, 1)
   318→   318→      ctx.fillRect(sx + 0.5, sy - 2, 1, 6)
   319→   319→    } else {
   320→   320→      ctx.fillRect(sx, sy, s > 1 ? 2 : 1, s > 1 ? 2 : 1)
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  // Atmospheric fog at bottom
   325→   325→  const fogGrad = ctx.createLinearGradient(0, vh * 0.8, 0, vh)
   326→   326→  fogGrad.addColorStop(0, 'rgba(10,15,40,0)')
   327→   327→  fogGrad.addColorStop(1, 'rgba(10,15,40,0.3)')
   328→   328→  ctx.fillStyle = fogGrad
   329→   329→  ctx.fillRect(0, vh * 0.8, vw, vh * 0.2)
   330→   330→}

   332→   332→/* ═══════════════════ Ground Drawing ═══════════════════ */
   333→   333→
   334→   334→function drawGround(ctx: CanvasRenderingContext2D, vw: number, vh: number, camX: number, camY: number) {
   335→   335→  // Grass patches with texture
   336→   336→  for (const g of GRASS) {
   337→   337→    const gx = (g.x - camX) | 0
   338→   338→    const gy = (g.y - camY) | 0
   339→   339→    if (gx < -16 || gx > vw + 16 || gy < -16 || gy > vh + 16) continue
   340→   340→    ctx.fillStyle = GRASS_SHADES[g.s]
   341→   341→    ctx.fillRect(gx, gy, g.w, g.h)
   342→   342→    // Grass blade highlights
   343→   343→    if (g.s % 2 === 0 && g.w > 5) {
   344→   344→      ctx.fillStyle = 'rgba(0,229,255,0.025)'
   345→   345→      ctx.fillRect(gx + 1, gy, 1, g.h)
   346→   346→    }
   347→   347→  }
   348→   348→
   349→   349→  // Paths with cobblestone texture
   350→   350→  for (const p of PATHS) {
   351→   351→    const px = (p.x - camX) | 0
   352→   352→    const py = (p.y - camY) | 0
   353→   353→    if (px > vw + p.w || px + p.w < 0 || py > vh + p.h || py + p.h < 0) continue
   354→   354→    // Base path
   355→   355→    ctx.fillStyle = '#2c2c54'
   356→   356→    ctx.fillRect(px, py, p.w, p.h)
   357→   357→    // Top edge highlight
   358→   358→    ctx.fillStyle = '#3d3d6b'
   359→   359→    ctx.fillRect(px, py, p.w, 2)
   360→   360→    // Bottom edge shadow
   361→   361→    ctx.fillStyle = 'rgba(0,0,0,0.35)'
   362→   362→    ctx.fillRect(px, py + p.h - 2, p.w, 2)
   363→   363→    // Cobblestone pattern
   364→   364→    ctx.fillStyle = 'rgba(61,61,107,0.35)'
   365→   365→    for (let cx = px + 5; cx < px + p.w - 5; cx += 10) {
   366→   366→      for (let cy = py + 3; cy < py + p.h - 3; cy += 6) {
   367→   367→        ctx.fillRect(cx + (cy % 2) * 5, cy, 7, 4)
   368→   368→      }
   369→   369→    }
   370→   370→    // Path glow lines
   371→   371→    ctx.fillStyle = 'rgba(0,229,255,0.05)'
   372→   372→    ctx.fillRect(px + 3, py + 2, p.w - 6, 1)
   373→   373→    ctx.fillRect(px + 3, py + p.h - 3, p.w - 6, 1)

   842→   842→  ctx.stroke()
   843→   843→}

  1055→  1055→    ctx.fill()
  1056→  1056→

  1060→  1060→    ctx.shadowBlur = 10

  1062→  1062→    ctx.lineWidth = 2
  1063→  1063→    roundRect(ctx, bubX, bubY, bw, bh, 6)

  1065→  1065→    ctx.restore()
  1066→  1066→

  1071→  1071→    ctx.fillStyle = bubGrad
  1072→  1072→    roundRect(ctx, bubX, bubY, bw, bh, 6)
  1073→  1073→    ctx.fill()
  1074→  1074→
  1075→  1075→    // Inner border
  1076→  1076→    ctx.strokeStyle = 'rgba(255,255,255,0.1)'
  1077→  1077→    ctx.lineWidth = 1

  1080→  1080→
  1081→  1081→    // Text with outline (CLEAR, BOLD)
  1082→  1082→    ctx.textAlign = 'center'
  1083→  1083→    ctx.textBaseline = 'middle'
  1084→  1084→    drawStrokedText(ctx, text, nx, bubY + bh / 2 + 1, 3, 'rgba(0,0,0,0.9)', '#FFFFFF')
  1085→  1085→
  1086→  1086→    ctx.globalAlpha = 1
  1087→  1087→  }
  1088→  1088→}

  1285→  1285→function drawPrompt(ctx: CanvasRenderingContext2D, building: BuildingData | null, camX: number, camY: number, isMobile: boolean, time: number) {
  1286→  1286→  if (!building) return
  1287→  1287→

  1289→  1289→  const by = (building.y + building.h + 20 - camY) | 0

  1317→  1317→  pGrad.addColorStop(0, 'rgba(0,229,255,0.1)')

  1327→  1327→
  1328→  1328→  ctx.globalAlpha = 1
